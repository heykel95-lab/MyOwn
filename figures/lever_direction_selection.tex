% Two views of one angular offset, for Section 2.7.2. The figure exists to keep
% two kinds of object apart, because they wear almost the same name:
%
%   theta_{t_1}, theta_{t_2}   two scalars, the first-order angular components
%                              about the surface tangents. Panel (a) shows the
%                              first as an angle between the tool face and the
%                              surface.
%   theta_{t_1} t_1, ...       the two vectors those scalars scale. Panel (b)
%                              draws them along the tangents, where they add to
%                              the angular-offset vector.
%
% An earlier version labelled the axis arrows of panel (b) theta_{t_1} and
% theta_{t_2}, so one name stood for an arc in one panel and an arrow in the
% other. The arrows carry the products now, which is what they are, and the
% offset vector is set bold to separate it from its scalar components.
%
% Panel (b) is also where the two components come from: the offset, its
% magnitude, the direction angle alpha and the right triangle closing on the
% tangents are all drawn, so cos and sin are visible as the projection they
% describe rather than only stated in the text below.
%
% Panel (a) looks along t_1, drawn out of the page as a dotted circle, so a
% positive rotation runs counter-clockwise, matching rotation_sign_convention.tex.
%
% Uses only tikz + arrows.meta, which config/packages.tex loads.
%
% Colour follows compliance_lever_moment.tex and case_c_direction_rule.tex: the
% surface and the tool are black with a light fill, red carries the angular
% offset and everything resolved from it, blue the compliance-centre
% displacement. Solid lines are quantities; the projections are construction and
% are dashed and thinner.
\begin{tikzpicture}[
    scale=1.15,
    font=\small,
    ax/.style={-{Latex[length=1.8mm]}, black},
    off/.style={-{Latex[length=2.2mm] Latex[length=2.2mm]}, thick, red},
    cmp/.style={-{Latex[length=1.6mm]}, red, line width=0.7pt},
    lev/.style={-{Latex[length=2.2mm]}, thick, blue},
    arc/.style={-{Latex[length=1.6mm]}, red},
    surf/.style={draw=black, fill=black!8},
    tool/.style={draw=black, fill=black!4},
    prj/.style={densely dashed, black, line width=0.7pt},
    lbl/.style={font=\footnotesize, inner sep=1.8pt},
    pt/.style={circle, fill=black, inner sep=0pt, minimum size=3.0pt},
    cpt/.style={circle, fill=blue, inner sep=0pt, minimum size=3.0pt},
  ]

% ============ (a) the component about t_1, seen as an angle =================
\begin{scope}[shift={(-4.15,0.55)}, scale=1.15]
  \fill[surf] (-1.50,-0.22) rectangle (1.50,0);
  \draw[black] (-1.50,0) -- (1.50,0);
  \draw[ax] (-1.34,0) -- (-1.34,0.82) node[lbl, above] {$n_s$};

  \begin{scope}[shift={(-0.72,0)}, rotate=16]
    \draw[tool] (0,0) rectangle (1.80,0.22);
  \end{scope}

  \draw[black, fill=white] (-0.72,0) circle (0.13);
  \fill[black] (-0.72,0) circle (0.04);
  \node[lbl, anchor=south east] at (-0.80,0.06) {$t_1$};

  \draw[arc] ([shift=(0:0.82)]-0.72,0) arc (0:16:0.82);
  \node[lbl, red] at ([shift=(8:1.07)]-0.72,0) {$\theta_{t_1}$};
\end{scope}

\node[lbl] at (-4.15,-1.75) {(a) Angle about $t_1$};

% ============ (b) the offset resolved in the tangent plane ==================
\begin{scope}
  \draw[ax] (0.36,0) -- (2.05,0);
  \node[lbl, anchor=west]  at (2.22,0) {$t_1$};
  \draw[ax] (0,0.36) -- (0,1.85);
  \node[lbl, anchor=south] at (0,1.98) {$t_2$};

  \draw[black, fill=white] (-1.32,-0.95) circle (0.13);
  \fill[black] (-1.32,-0.95) circle (0.04);
  \node[lbl, anchor=east] at (-1.52,-0.95) {$n_s$};

  % ---- the two component vectors, on the tangents they lie along ----------
  \draw[cmp] (0,0) -- (1.474,0);
  \draw[cmp] (0,0) -- (0,1.032);
  \node[lbl, red, anchor=north] at (1.474,-0.10) {$\theta_{t_1}t_1$};
  \node[lbl, red, anchor=north west] at (0.13,0.94) {$\theta_{t_2}t_2$};

  % ---- their sum, with the right triangle the two components close ---------
  \draw[prj] (1.474,0) -- (35:1.80);
  \draw[prj] (0,1.032) -- (35:1.80);
  \draw[black, thin] (1.314,0) -- (1.314,0.16) -- (1.474,0.16);
  \draw[off] (0,0) -- (35:1.80);
  \node[lbl, red, anchor=west] at (35:2.08) {$\boldsymbol{\theta}$};
  \draw[arc] (0.52,0) arc (0:35:0.52);
  \node[lbl, red] at (17.5:0.74) {$\alpha$};
  \node[lbl, red] at (1.12,0.45) {$\lVert\boldsymbol{\theta}\rVert$};

  % ---- what the triangle states, in the two relations it stands for -------
  \node[lbl, red, align=left] at (1.30,-1.12)
    {$\theta_{t_1}=\lVert\boldsymbol{\theta}\rVert\cos\alpha$\\
     $\theta_{t_2}=\lVert\boldsymbol{\theta}\rVert\sin\alpha$};

  % ---- the displacement selected perpendicular to it ----------------------
  \draw[lev] (0,0) -- (125:1.60);
  \node[cpt] at (125:1.60) {};
  \node[lbl, blue, anchor=east] at (-1.05,1.23) {$p_c$};
  \node[lbl, blue] at (125:1.98) {$r_{c,t}$};
  \node[pt] at (0,0) {};
  \node[lbl, anchor=north west] at (0.10,-0.13) {TCP};

  % ---- right angle between the offset and the displacement ----------------
  % Rotated by the offset direction, the first of the two, so the square spans
  % the quadrant between them.
  \begin{scope}[rotate=35]
    \draw[black, thin] (0.34,0) -- (0.34,0.34) -- (0,0.34);
  \end{scope}

  \node[lbl] at (0.35,-1.75) {(b) Offset resolved in the tangent plane};
\end{scope}

\end{tikzpicture}
