% Three views of one angular offset, for Section 2.7.2. Each panel answers one
% question, because an earlier two-panel version carried the offset, its
% components, the direction angle, the TCP, p_c and the displacement in a single
% drawing and was hard to read:
%
%   (a) what the offset is physically   an angle between the tool face and the
%                                       surface, seen in the elevation along t_1
%   (b) what the offset vector is       one arrow in the tangent plane, resolved
%                                       into two coordinates by projection
%   (c) how the displacement follows    the tangential displacement drawn
%                                       perpendicular to that arrow
%
% Panel (b) draws the components as dashed projections and labels the segments
% they mark, not as arrows: an arrow on each axis reads as a second and third
% rotation, when there is one small-angle vector with two coordinates. The
% vector carries one arrowhead like any other: a double head was tried, as the
% notation for a rotation vector, and dropped once the symbol was set bold. The
% offset itself is bold, matching the prose, and its components are not.
%
% Panel (a) looks along t_1, drawn out of the page as a dotted circle, so a
% positive rotation runs counter-clockwise, matching rotation_sign_convention.tex.
% Panels (b) and (c) look along n_s, which is drawn out of the page in (c) where
% the cross product needs it.
%
% Uses only tikz + arrows.meta, which config/packages.tex loads.
%
% Colour follows compliance_lever_moment.tex and case_c_direction_rule.tex: the
% surface and the tool are black with a light fill, red carries the angular
% offset and everything resolved from it, blue the compliance-centre
% displacement. Solid lines are quantities; the projections are construction and
% are dashed and thinner.
\begin{tikzpicture}[
    scale=1.1,
    font=\small,
    ax/.style={-{Latex[length=1.8mm]}, black},
    off/.style={-{Latex[length=2.2mm]}, thick, red},
    lev/.style={-{Latex[length=2.2mm]}, thick, blue},
    arc/.style={-{Latex[length=1.6mm]}, red},
    surf/.style={draw=black, fill=black!8},
    tool/.style={draw=black, fill=black!4},
    prj/.style={densely dashed, black, line width=0.7pt},
    lbl/.style={font=\footnotesize, inner sep=1.8pt},
    pt/.style={circle, fill=black, inner sep=0pt, minimum size=3.0pt},
    cpt/.style={circle, fill=blue, inner sep=0pt, minimum size=3.0pt},
  ]

% ============ (a) the offset as an angle at the contact =====================
\begin{scope}[shift={(-5.60,0.30)}]
  \fill[surf] (-1.45,-0.22) rectangle (1.45,0);
  \draw[black] (-1.45,0) -- (1.45,0);
  \draw[ax] (-1.30,0) -- (-1.30,0.80) node[lbl, above] {$n_s$};

  \begin{scope}[shift={(-0.70,0)}, rotate=16]
    \draw[tool] (0,0) rectangle (1.75,0.21);
  \end{scope}

  \draw[black, fill=white] (-0.70,0) circle (0.13);
  \fill[black] (-0.70,0) circle (0.04);
  \node[lbl, anchor=south east] at (-0.78,0.06) {$t_1$};

  \draw[arc] ([shift=(0:0.80)]-0.70,0) arc (0:16:0.80);
  \node[lbl, red] at ([shift=(8:1.05)]-0.70,0) {$a_{t_1}$};
\end{scope}
\node[lbl] at (-5.60,-1.55) {(a) Angle about $t_1$};

% ============ (b) the offset vector and its two coordinates =================
\begin{scope}[shift={(-1.55,0)}]
  \draw[ax] (0,0) -- (1.80,0);
  \node[lbl, anchor=west]  at (1.94,0) {$t_1$};
  \draw[ax] (0,0) -- (0,1.60);
  \node[lbl, anchor=south] at (0,1.72) {$t_2$};

  \draw[prj] (35:1.55) -- (1.270,0);
  \draw[prj] (35:1.55) -- (0,0.889);
  \draw[off] (0,0) -- (35:1.55);
  \node[lbl, red, anchor=west] at (35:1.80) {$\mathbf{a}$};
  % Set clear of the shaft on the free side, not merely anchored above it: at
  % this size an anchored label still printed across the arrow.
  \node[lbl, red, rotate=35] at ([shift=(125:0.26)]35:0.82) {$\lVert\mathbf{a}\rVert$};

  \draw[arc] (0.42,0) arc (0:35:0.42);
  \node[lbl, red] at (17.5:0.62) {$\alpha$};

  \node[lbl, red, anchor=north] at (0.635,-0.08) {$a_{t_1}$};
  \node[lbl, red, anchor=east] at (-0.08,0.445) {$a_{t_2}$};
\end{scope}
\node[lbl] at (-0.70,-1.55) {(b) Offset vector in the tangent plane};

% ============ (c) the displacement selected from that direction =============
\begin{scope}[shift={(3.95,0)}]
  \draw[ax] (0,0) -- (1.45,0);
  \node[lbl, anchor=west]  at (1.58,0) {$t_1$};
  \draw[ax] (0,0) -- (0,1.45);
  \node[lbl, anchor=south] at (0,1.57) {$t_2$};

  \draw[black, fill=white] (-1.05,-0.75) circle (0.13);
  \fill[black] (-1.05,-0.75) circle (0.04);
  \node[lbl, anchor=east] at (-1.25,-0.75) {$n_s$};

  \draw[off] (0,0) -- (35:1.40);
  \node[lbl, red, anchor=west] at (35:1.62) {$\mathbf{a}$};

  \draw[lev] (0,0) -- (125:1.40);
  \node[cpt] at (125:1.40) {};
  \node[lbl, blue, anchor=east] at (-0.92,1.07) {$p_c$};
  \node[lbl, blue] at (125:1.76) {$r_{c,t}$};
  \node[pt] at (0,0) {};
  \node[lbl, anchor=north west] at (0.08,-0.10) {TCP};

  % The square spans the quadrant between the two, so it is rotated by the
  % offset direction, the first of them.
  \begin{scope}[rotate=35]
    \draw[black, thin] (0.30,0) -- (0.30,0.30) -- (0,0.30);
  \end{scope}
\end{scope}
\node[lbl] at (3.95,-1.55) {(c) Selected displacement};

\end{tikzpicture}
