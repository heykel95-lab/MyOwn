% Classical Cartesian feedback loop with additive model and null-space terms.
% Blocks name operations; every arrow names the transmitted signal.
\begin{tikzpicture}[
    scale=0.98,
    font=\small\setstretch{1.0},
    blk/.style={draw=black, line width=0.75pt, minimum height=1.10cm,
                inner xsep=1.2mm, inner ysep=0.8mm, align=center,
                outer sep=0pt, font=\scriptsize\setstretch{1.0}},
    main/.style={blk, minimum height=1.25cm},
    sig/.style={-{Latex[length=2mm]}, thick, black},
    info/.style={-{Latex[length=1.8mm]}, line width=0.7pt, black},
    lbl/.style={font=\scriptsize\setstretch{1.0}, inner sep=1.2pt},
  ]

  % Dominant reference-to-robot path.
  \node[main] (error) at (-6.20,0.00) {Cartesian Error};
  \node[main] (controller) at (-2.50,0.00) {Impedance Controller};
  \node[main] (mapping) at (0.45,0.00) {$J^\top$ Mapping};
  \node[main] (sum) at (3.05,0.00) {Torque Sum};
  \node[main] (robot) at (5.75,0.00) {Robot / Joint Motors};

  \draw[sig] (error.east) -- node[lbl, above=1.5pt] {$e_p,\ e_R$} (controller.west);
  \draw[sig] (controller.east) -- node[lbl, above=1.5pt] {$F$} (mapping.west);
  \draw[sig] (mapping.east) -- node[lbl, above=1.5pt] {$\tau_{\mathrm{cart}}$} (sum.west);
  \draw[sig] (sum.east) -- node[lbl, above=1.5pt] {$\tau_{\mathrm{cmd}}$} (robot.west);

  % Desired Cartesian reference enters the comparison from above.
  \node[blk] (reference) at (-6.20,2.15) {Desired Cartesian Reference};
  \draw[sig] (reference.south) --
      node[lbl, right] {$p_d,\ R_d,\ \dot p_d$} (error.north);

  % Robot-model and null-space terms remain secondary to the main path.
  \node[blk] (model) at (0.45,2.15) {Robot Model};
  \node[blk] (null) at (4.35,2.15) {Null-Space Term};

  \draw[info] (model.south) -- node[lbl, right] {$J(q)$} (mapping.north);
  \draw[info] (model.east) --
      node[lbl, above] {$q,\ \dot q,\ J(q)$} (null.west);

  \coordinate (coriolislevel) at (1.45,1.02);
  \coordinate (suminleft) at ($(sum.north west)!0.25!(sum.north east)$);
  \draw[info] (model.south east) -- (model.south east |- coriolislevel) --
      node[lbl, above] {$\tau_c(q,\dot q)$} (suminleft |- coriolislevel) --
      (suminleft);

  \coordinate (suminright) at ($(sum.north west)!0.72!(sum.north east)$);
  \coordinate (nulllevel) at (4.35,1.28);
  \draw[info] (null.south) --
      node[lbl, right, pos=0.40] {$\tau_{\mathrm{null}}$} (nulllevel) --
      (suminright |- nulllevel) -- (suminright);

  % The robot-state interface supplies the physical feedback and model input.
  \node[blk] (measured) at (5.75,-2.15) {Measured Robot State};
  \draw[sig] (robot.south) --
      node[lbl, left] {Joint motion / sensing} (measured.north);

  \coordinate (feedbacklevel) at (0.00,-3.65);
  \coordinate (errorin) at ($(error.south west)!0.35!(error.south east)$);
  \draw[sig] (measured.south west) -- (measured.south west |- feedbacklevel) --
      node[lbl, above] {$p_{\mathrm{EE}},\ R_{\mathrm{EE}},\ \dot p_{\mathrm{EE}},\ \omega_{\mathrm{EE}}$}
      (errorin |- feedbacklevel) -- (errorin);

  \coordinate (stateout) at ($(measured.south east)!0.62!(measured.north east)$);
  \coordinate (statehigh) at (7.65,3.45);
  \draw[info] (stateout) -- (stateout -| statehigh) -- (statehigh) --
      node[lbl, above] {$q,\ \dot q$} (model.north |- statehigh) -- (model.north);

\end{tikzpicture}
