% Block diagram of the torque command assembled once per control cycle.
% Uses only tikz + arrows.meta/calc/positioning, which config/packages.tex loads.
% Routing keeps four horizontal runs at distinct heights so no signal crosses
% another: J above the chain, the gains and the state feed just below it, the
% null-space and Coriolis contributions below that, and the loop closure last.
\begin{tikzpicture}[
    font=\small,
    blk/.style={draw=gray!70!black, fill=white, rounded corners=2pt,
                minimum height=9mm, inner xsep=2.5mm, align=center},
    src/.style={draw=gray!60, fill=white, rounded corners=2pt,
                minimum height=9mm, inner xsep=2.5mm, align=center},
    sel/.style={draw=gray!70!black, fill=white, rounded corners=2pt,
                minimum height=9mm, inner xsep=2.5mm, align=center},
    sum/.style={draw=gray!70!black, fill=white, circle, minimum size=6mm,
                inner sep=0pt},
    sig/.style={-{Latex[length=2mm]}, thick, gray!55!black},
    lbl/.style={font=\footnotesize, inner sep=1.6pt},
  ]

% ------------------------------------------------------------ forward chain
\node[src] (state)  at ( 0.0,  0.0) {robot state\\[-2pt]\footnotesize $q,\ \dot q,\ T_{EE}$};
\node[sel] (phase)  at ( 3.5,  0.0) {phase machine\\[-2pt]\footnotesize $p_d,\ R_d$};
\node[blk] (err)    at ( 7.0,  0.0) {Cartesian error\\[-2pt]\footnotesize $e_p,\ e_R$};
\node[blk] (wrench) at (10.5,  0.0) {spring wrench\\[-2pt]\footnotesize $F=K\Delta x+D\Delta v$};
\node[blk] (jt)     at (13.8,  0.0) {\footnotesize $\tau_{\text{task}}=J^{\!\top}F$};

% ------------------------------------------------------------- lower inputs
\node[src] (model)  at ( 0.0, -2.1) {model\\[-2pt]\footnotesize $J,\ M(q),\ c(q,\dot q)$};
\node[sel] (null)   at ( 3.5, -2.1) {null-space term\\[-2pt]\footnotesize $N_\tau$: damping, $\sigma_{\min}$};
% The text width is fixed rather than left to the content, and the value is a
% compromise between two failures. Unconstrained, this box reaches past
% x = 13.5 and the jt-to-sum descent at x = 13.8 runs through it. Narrowed to
% 3.5 cm it instead grows tall, and at 4.6 cm it still wrapped to four lines
% and stood 2 cm high against 0.9 cm for every other box, leaving the M(q) run
% 0.16 cm below it. 5.2 cm wraps to three lines and keeps the right edge clear
% of the descent. Re-measure with \typeout on the node anchors before changing
% this value; the extents are not obvious from the source.
\node[sel, text width=5.2cm, align=center] (imp) at (10.5, -2.1)
  {impedance form\\[-2pt]\footnotesize decoupled \emph{or} point-shifted,
   inertia-scaled $D$};

% -------------------------------------------------------------- summation
\node[sum] (plus)   at (13.8, -3.95) {\large$+$};
\node[blk] (out)    at (16.6, -3.95) {joint torque\\[-2pt]\footnotesize $\tau_{\mathrm{cmd}}$};

% ------------------------------------------------------- chain, left to right
\draw[sig] (state)  -- (phase);
\draw[sig] (phase)  -- (err);
\draw[sig] (err)    -- (wrench);
\draw[sig] (wrench) -- (jt);
\draw[sig] (imp.north) -- (wrench.south);
\draw[sig] (plus) -- (out);
\draw[sig] (jt.south) -- (plus.north);
\draw[sig] (model) -- (null);

% run 1: J, above the chain
\draw[sig] (model.west) -- (-3.0,-2.1) -- (-3.0,1.45)
      -- node[lbl, above, pos=0.06] {$J$} (13.8,1.45) -- (jt.north);

% run 2: measured pose into the error, just under the chain
\draw[sig] (state.south) -- (0.0,-0.95) -- (7.0,-0.95) -- (err.south);

% run 3: the active gains select the impedance form. It enters from the west
% rather than the north, because the north edge already carries the link up to
% the spring wrench; sharing that edge put two opposed arrowheads on one
% segment and made the link cross this run. Its descent sits at x = 7.4, right
% of where run 2 ends at x = 7.0 and left of the box edge at x = 7.95.
\draw[sig] (phase.south) -- (3.5,-1.35) -- node[lbl, above, pos=0.5]
      {active gains} (7.4,-1.35) -- (7.4,-2.1) -- (imp.west);

% run 4: null-space torque into the sum from the left
\draw[sig] (null.south) -- (3.5,-3.95) -- node[lbl, above, pos=0.55]
      {$\tau_{\text{null}}$} (13.5,-3.95);

% M(q) has exactly one consumer: it sizes the damping, not the control law
% The measured box bottom is y = -3.09, so this run sits at -3.60 rather than
% -3.25; at -3.25 its arrowhead met the box after a 0.16 cm rise.
\draw[sig] (model.south) -- (0.0,-3.60) -- node[lbl, above, pos=0.62]
      {$M(q)$ sizes $D$} (10.5,-3.60) -- (imp.south);

% run 5: Coriolis compensation into the sum from below
\draw[sig] (model.south) -- (0.0,-5.15) -- node[lbl, below, pos=0.55]
      {$c(q,\dot q)$} (13.8,-5.15) -- (plus.south);

% run 6: the loop closes through the robot, around the left
\draw[sig] (out.south) -- (16.6,-5.95) -- node[lbl, below, pos=0.55]
      {robot, \(1\,\mathrm{kHz}\)} (-1.9,-5.95) -- (-1.9,0.0) -- (state.west);

\end{tikzpicture}
