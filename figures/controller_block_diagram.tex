% Functional torque-control architecture.
% Blocks name operations; every arrow names the transmitted signal.
\begin{tikzpicture}[
    scale=1.04,
    font=\small\setstretch{1.0},
    blk/.style={draw=black, thick, minimum height=1.0cm,
                inner xsep=1mm, align=center, outer sep=0pt},
    main/.style={blk, minimum height=1.55cm},
    sig/.style={-{Latex[length=2mm]}, thick, black},
    info/.style={-{Latex[length=2mm]}, line width=0.7pt, black},
    branch/.style={thick, black},
    lbl/.style={font=\footnotesize\setstretch{1.0}, inner sep=1pt},
  ]

  % Main left-to-right torque path.
  \node[main, text width=1.80cm] (error)  at (-6.20,0.00) {Cartesian Error};
  \node[main, text width=2.00cm] (wrench) at (-3.00,0.00)
      {Cartesian Impedance Wrench};
  \node[main, text width=1.65cm] (cart)   at (-0.25,0.00) {Cartesian Torque};
  \node[main, text width=1.40cm] (sum)    at ( 2.65,0.00) {Torque Sum};
  \node[main, text width=1.50cm] (motors) at ( 5.30,0.00) {Joint Motors};

  \draw[sig] (error) -- node[lbl, above] {$e_p,\ e_R$} (wrench);
  \draw[sig] (wrench) -- node[lbl, above] {$F$} (cart);
  \draw[sig] (cart) -- node[lbl, above] {$\tau_{\mathrm{cart}}$} (sum);
  \draw[sig] (sum) -- node[lbl, above] {$\tau_{\mathrm{cmd}}$} (motors);

  % State-selected references and gains.
  \node[blk, text width=2.60cm] (machine) at (-4.40,2.55)
      {Controller State};
  \coordinate (statesplit) at (-4.40,1.35);
  \draw[branch] (machine.south) -- (statesplit);
  \draw[sig] (statesplit) --
      node[lbl, above] {$p_d,\ R_d$} (-6.20,1.35) -- (error.north);
  \draw[sig] (statesplit) -- (-3.00,1.35) --
      node[lbl, right, pos=0.38] {$K,\ D,\ \dot p_d$} (wrench.north);

  % Joint sensing and the robot model remain distinct.
  \node[blk, text width=2.10cm] (feedback) at (-6.20,-2.45)
      {Joint Feedback};
  \node[blk, text width=2.10cm] (model) at (-0.25,-2.45)
      {Robot Model};

  \draw[sig] (feedback.east) -- node[lbl, above] {$q,\ \dot q$} (model.west);
  \draw[sig] (feedback.north) -- node[lbl, left] {$T_{EE}$} (error.south);
  \coordinate (velocitylevel) at (0.00,-1.25);
  \draw[sig] (feedback.north east) -- (feedback.north east |- velocitylevel) --
      node[lbl, above] {$\dot p,\ \omega$} (wrench.south |- velocitylevel) --
      (wrench.south);

  \draw[sig] (model.north) -- node[lbl, right] {$J(q)$} (cart.south);
  \draw[sig] (model.east) -- (2.65,-2.45) --
      node[lbl, right, pos=0.72] {$\tau_c(q,\dot q)$} (sum.south);

  % The selectable null-space contribution is a short secondary input.
  \node[blk, text width=2.10cm] (null) at (2.65,2.55)
      {Null-Space Term};
  \draw[info] (feedback.west) -- (-7.65,-2.45) -- (-7.65,3.55)
      -- node[lbl, above] {Robot state/model} (2.65,3.55) -- (null.north);
  \draw[sig] (null.south) -- node[lbl, right] {$\tau_{\mathrm{null}}$} (sum.north);

  % One physical loop closes the diagram.
  \draw[sig] (motors.south) -- (5.30,-3.85) --
      node[lbl, below, pos=0.50] {Joint motion} (-6.20,-3.85) -- (feedback.south);

\end{tikzpicture}
