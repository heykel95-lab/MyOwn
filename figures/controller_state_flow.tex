% Program-level controller states and their transition conditions.
% Condition labels sit beside the routes so no text interrupts a line.
\begin{tikzpicture}[
    scale=1.0,
    font=\footnotesize\setstretch{1.0},
    state/.style={draw=black, thick, text width=3.15cm, align=center,
                  inner sep=4pt, minimum height=1.0cm},
    stopstate/.style={draw=red, text=red, thick, text width=3.15cm,
                      align=center, inner sep=4pt, minimum height=1.0cm},
    flow/.style={-{Latex[length=2mm]}, thick, black},
    stopflow/.style={-{Latex[length=2mm]}, thick, red},
    rail/.style={thick, red},
    cond/.style={font=\scriptsize\setstretch{1.0}, align=left, inner sep=0pt},
    stopcond/.style={font=\scriptsize\setstretch{1.0}, align=center,
                     text=red, inner sep=0pt},
  ]

  \node[state] (recovery)  at (0.00, 0.00) {Robot Recovery};
  \node[state] (selection) at (0.00,-1.75) {Controller Selection};

  \node[state] (surface)  at (-4.60,-4.40) {Contact Sequence};
  \node[state] (pose)     at ( 0.00,-4.40) {Cartesian Pose Hold};
  \node[state] (guidance) at ( 4.60,-4.40) {Manual Guidance};
  \node[stopstate] (stop) at ( 0.00,-7.90) {Stop};

  \draw[flow] (recovery) -- (selection);
  \node[cond, anchor=west] at (0.25,-0.88) {if recovery succeeds};

  \coordinate (split) at (0.00,-2.90);
  \draw[flow] (selection) -- (split);
  \draw[flow] (split) -- (-4.60,-2.90) -- (surface);
  \draw[flow] (split) -- (pose);
  \draw[flow] (split) -- (4.60,-2.90) -- (guidance);
  \node[cond, anchor=south west] at (-4.30,-2.60)
      {if operator types \texttt{s}};
  \node[cond, anchor=west] at (0.25,-3.60)
      {if operator types \texttt{h}};
  \node[cond, anchor=south east, align=right] at (4.30,-2.60)
      {if operator types \texttt{g}};

  % One shared stop rail for all active controller states.
  \coordinate (railleft) at (-4.60,-5.80);
  \coordinate (railright) at (4.60,-5.80);
  \draw[stopflow] (surface) -- (railleft);
  \draw[stopflow] (pose) -- (0.00,-5.80);
  \draw[stopflow] (guidance) -- (railright);
  \draw[rail] (railleft) -- (railright);
  \draw[stopflow] (0.00,-5.80) -- (stop);
  \node[stopcond, anchor=west, align=left] at (0.35,-6.70)
      {if stop requested \(\lor\) robot error/reflex};

  % Recovery can fail before a controller state is selected.
  \draw[stopflow] (recovery.west) -- (-6.85,0.00) -- (-6.85,-7.90)
      -- (stop.west);
  \node[stopcond, anchor=south] at (-5.05,0.20) {if recovery fails};

\end{tikzpicture}
