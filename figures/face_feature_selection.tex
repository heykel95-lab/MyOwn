% Rectangular grinding face in the end-effector frame, and the three outcomes of
% the leading-feature comparison along the descent direction.
% Uses only tikz + arrows.meta/calc/positioning, which config/packages.tex loads.
\begin{tikzpicture}[
    scale=1.30,
    font=\footnotesize,
    face/.style={draw=gray!70!black, fill=white},
    vec/.style={-{Latex[length=1.8mm]}, gray!55!black},
    sel/.style={draw=gray!70!black, fill=gray!25},
    lbl/.style={font=\footnotesize, inner sep=1.6pt},
  ]

  % ---- calibrated face, half-width along x_EE and half-length along y_EE
  \draw[face] (0,0) rectangle (3.2,1.6);

  % ---- the three outcomes of the comparison along the descent direction
