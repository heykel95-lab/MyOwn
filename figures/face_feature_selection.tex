% Rectangular tool face in the end-effector frame: the calibrated centre
% offset, the half-width and half-length vectors, and the four corners.
% Uses only tikz + arrows.meta/calc/positioning, which config/packages.tex loads.
\begin{tikzpicture}[
    scale=1.30,
    font=\footnotesize,
    face/.style={draw=black},
    vec/.style={-{Latex[length=1.8mm]}, black},
    lbl/.style={font=\footnotesize, inner sep=1.6pt},
  ]

  % ---- calibrated face, half-width along x_EE and half-length along y_EE
  \draw[face] (0,0) rectangle (3.2,1.6);
  \fill (1.6,0.8) circle (1.2pt);
  \node[lbl, below left] at (1.52,0.72) {$r_{\mathrm{face,EE}}$};

  \draw[vec] (1.6,0.8) -- (3.2,0.8);
  \node[lbl, below] at (2.45,0.66) {$r_{\mathrm{width,EE}}$};
  \draw[vec] (1.6,0.8) -- (1.6,1.6);
  \node[lbl, right] at (1.70,1.30) {$r_{\mathrm{length,EE}}$};

  \foreach \x/\y in {0/0, 3.2/0, 0/1.6, 3.2/1.6} {\fill (\x,\y) circle (1.4pt);}
