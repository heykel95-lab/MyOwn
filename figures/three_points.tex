% The three points the contact sequence keeps apart: the selected grinding-face
% point, the tool centre point, and the virtual centre of compliance.
% The compliance centre is drawn clear of the tool so the two offsets and their
% sum each get their own space; no label sits on a line or inside a node.
\begin{tikzpicture}[
    font=\small,
    tool/.style={draw=gray!70!black, fill=white},
    off/.style={-{Latex[length=2mm]}, gray!55!black, thick},
    sum/.style={-{Latex[length=2mm]}, gray!70!black, thick, densely dashed},
    lbl/.style={font=\small, inner sep=2pt},
  ]

  % ---- surface and the tool face resting above it
  \draw[gray!70!black] (0,0) -- (7.2,0);
  \node[lbl, anchor=west] at (5.9,-0.32) {surface};
  \draw[tool] (1.0,0.5) rectangle (4.2,1.1);

  % ---- the three points
  \fill (1.3,0.5) circle (1.8pt);
  \node[lbl, below] at (1.3,0.42) {$p_g$};

  \fill (2.6,0.8) circle (1.8pt);
  \node[lbl, above] at (2.6,1.18) {$p_{\mathrm{TCP}}$};

  \fill (5.8,2.2) circle (1.8pt);
  \node[lbl, right] at (5.95,2.2) {$p_c$};

  % ---- offset fixed by the tool and contact geometry
  \draw[off] (2.6,0.8) -- (1.3,0.5);
  \node[lbl, below] at (1.95,0.5) {$p_g-p_{\mathrm{TCP}}$};

  % ---- the selected controller parameter
  \draw[off] (5.8,2.2) -- (2.6,0.8);
  \node[lbl, above right] at (4.3,1.45) {$r_c=p_{\mathrm{TCP}}-p_c$};

  % ---- their sum, routed clear of both arrows
  \draw[sum] (5.8,2.2) .. controls (4.0,3.3) and (2.2,2.9) .. (1.3,0.5);
  \node[lbl, above] at (3.3,3.02) {$p_g-p_c$};

\end{tikzpicture}
