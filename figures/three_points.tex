% The three points the contact sequence keeps apart: the selected tool point,
% the tool centre point, and the virtual centre of compliance. Every label uses
% the same lbl style, so no symbol is set larger or heavier than its neighbours.
% The compliance centre is drawn clear of the tool so the two offsets and their
% sum each get their own space; no label sits on a line or inside a node.
\begin{tikzpicture}[
    scale=1.60,
    font=\small,
    tool/.style={draw=gray!70!black, fill=white},
    off/.style={-{Latex[length=2mm]}, gray!55!black, thick},
    sum/.style={-{Latex[length=2mm]}, gray!70!black, thick, densely dashed},
    lbl/.style={font=\small, inner sep=2pt},
  ]

  % ---- surface and the tool face resting above it
  \draw[gray!70!black] (0,0) -- (7.2,0);
  \node[lbl, anchor=west] at (5.9,-0.32) {surface};
  \draw[tool] (1.0,0.95) rectangle (4.2,1.55);

  % ---- the three points
  \fill (1.3,0.95) circle (1.8pt);
  \node[lbl, below left] at (1.22,0.86) {$p_T$};

  \fill (2.6,1.25) circle (1.8pt);
  \node[lbl, above] at (2.6,1.64) {$p_{\mathrm{TCP}}$};

  \fill (5.8,2.2) circle (1.8pt);
  \node[lbl, right] at (5.95,2.2) {$p_c$};

  % ---- offset fixed by the tool and contact geometry
  \draw[off] (2.6,1.25) -- (1.3,0.95);
  \node[lbl, below] at (2.30,0.42) {$r_T=p_T-p_{\mathrm{TCP}}$};

  % ---- the selected controller parameter
  \draw[off] (2.6,1.25) -- (5.8,2.2);
  \node[lbl, below right] at (4.45,1.58) {$r_c=p_c-p_{\mathrm{TCP}}$};

  % ---- their sum, routed clear of both arrows
  \draw[sum] (5.8,2.2) .. controls (4.0,3.4) and (2.2,3.0) .. (1.3,0.95);
  \node[lbl, above] at (3.3,3.18) {$p_T-p_c=r_T-r_c$};

\end{tikzpicture}
