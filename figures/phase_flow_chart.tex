% Surface-contact phase sequence: the four phases and the unboxed clearance
% transition between approach and set-up. Operator gates, hold modes, source
% identifiers and gain groups are deliberately left out of this overview.
% Uses only tikz + arrows.meta, which config/packages.tex loads.
\begin{tikzpicture}[
    scale=1.0,
    font=\footnotesize,
    phase/.style={draw=black, thick, text width=4.0cm, align=center,
                  inner sep=4pt, minimum height=1.0cm},
    evt/.style={align=center, inner sep=3pt},
    flow/.style={-{Latex[length=2mm]}, thick, black},
  ]

  % ---- the chain, top to bottom on x = 0
  \node[phase] (orient) at (0, 0.00) {Tool orientation};
  \node[phase] (appr)   at (0,-1.80) {Surface approach};
  \node[evt]   (clear)  at (0,-3.45) {Clearance transition};
  \node[phase] (setup)  at (0,-5.10) {Contact set-up};
  \node[phase] (grind)  at (0,-6.90) {Grinding};

  % ---- flow
  \draw[flow] (orient) -- (appr);
  \draw[flow] (appr)   -- (clear);
  \draw[flow] (clear)  -- (setup);
  \draw[flow] (setup)  -- (grind);

\end{tikzpicture}
