% Listing design after https://tex.stackexchange.com/q/305566, posted by
% william007 and modified by the community, retrieved 2026-08-06, CC BY-SA 3.0.
% A rule at the left instead of a box, no background tint, bold keywords and
% grey comments. The fonts of that answer are not adopted: pxfonts would change
% every equation in the thesis, not only the listings.
\lstdefinestyle{cppstyle}{
  language=C++,
  basicstyle=\ttfamily\scriptsize\setstretch{0.9},
  identifierstyle=\color{black},
  keywordstyle=\bfseries,
  commentstyle=\color{darkgray}\ttfamily,
  stringstyle=\color{black}\ttfamily,
  frame=l,
  framerule=0.6pt,
  rulecolor=\color{black},
  framesep=6pt,
  numbers=left,
  stepnumber=1,
  firstnumber=1,
  numberfirstline=true,
  numberstyle=\tiny\color{darkgray},
  numbersep=8pt,
  breaklines=true,
  breakatwhitespace=true,
  captionpos=t,
  columns=fullflexible,
  keepspaces=true,
  tabsize=2,
  showtabs=false,
  showspaces=false,
  showstringspaces=false,
  extendedchars=true,
  xleftmargin=0.75cm,
  xrightmargin=0.2em,
  aboveskip=0.8em,
  belowskip=0.6em
}
