\section{Reference Frames}
\label{sec:reference_frames}

\Revised{The following right-handed coordinate frames are used in the theoretical
formulation and the controller
\mbox{\cite{Siciliano2009,Craig2005}}.}

\begin{itemize}
  \item \Revised{\mbox{\textbf{Base frame \mbox{\(\{0\}\)}:}}
        The base frame is fixed to the robot mounting surface. Its origin is
        the robot base reference point used by the kinematic model. For the
        upright Panda configuration used in this thesis, the \mbox{\(z_0\)}-axis is
        normal to the mounting surface, while the \mbox{\(x_0\)}- and \mbox{\(y_0\)}-axes
        lie in the mounting plane. Unless stated otherwise, Cartesian
        positions, directions, velocities, forces, and moments are expressed
        in this frame.}

  \item \Revised{\mbox{\textbf{Joint frames
        \mbox{\(\{1\},\{2\},\ldots,\{n\}\)}:}}
        These frames are attached to the successive links of the kinematic
        chain according to the Denavit--Hartenberg convention
        \mbox{\cite{Denavit1955,Craig2005}}. Joint \mbox{\(i\)} rotates about
        \mbox{\(z_{i-1}\)}, and \mbox{\(p_{i-1}\)} denotes a point on this axis. The joint
        frames define the transformations between consecutive links and the
        indexing used in the forward kinematics and geometric Jacobian.}

  \item \Revised{\mbox{\textbf{End-effector frame
        \mbox{\(\{\mathrm{EE}\}\)}:}}
        The end-effector frame is attached to the TCP and moves with the tool.
        Its position \mbox{\(p_{\mathrm{EE}}\)} and orientation
        \mbox{\(R_{\mathrm{EE}}\)}, both expressed relative to the base frame,
        define the current Cartesian pose of the robot. The axes
        \mbox{\(x_{\mathrm{EE}}\)}, \mbox{\(y_{\mathrm{EE}}\)}, and
        \mbox{\(z_{\mathrm{EE}}\)} describe the current tool orientation. The
        configured tool approach axis is defined in this frame and is
        transformed to the base frame using \mbox{\(R_{\mathrm{EE}}\)}.}

  \item \Revised{\mbox{\textbf{Desired end-effector frame \mbox{\(\{d\}\)}:}}
        The desired frame represents the commanded TCP pose. Its origin is
        \mbox{\(p_d\)}, and its orientation is \mbox{\(R_d\)}, both expressed relative to
        the base frame. The difference between the desired and current
        end-effector frames provides the translational and rotational errors
        used by the impedance controller.}

  \item \Revised{\mbox{\textbf{Surface task frame \mbox{\(\{S\}\)}:}}
        The surface task frame is tied to the configured surface rather than
        to the moving tool. Its origin is placed at the configured surface
        point \mbox{\(p_s\)}, and its orientation is defined by the two surface
        tangents \mbox{\(t_1\)} and \mbox{\(t_2\)} and the surface normal \mbox{\(n_s\)}:}

        \begin{equation}
        R_{\mathrm{surface}}
        =
        \begin{bmatrix}
          t_1 & t_2 & n_s
        \end{bmatrix}.
        \label{eq:surface_frame_overview}
        \end{equation}

\Revised{        The directions \mbox{\(t_1\)} and \mbox{\(t_2\)} lie in the surface plane and are
        mutually orthogonal, while \mbox{\(n_s\)} is normal to the surface. This
        frame is used to describe the commanded tilt and to assign stiffness
        and damping values according to the two tangential directions and the
        normal direction. For the contact task,}

        \begin{equation}
        R_{\mathrm{task}}=R_{\mathrm{surface}}.
        \end{equation}
\end{itemize}

\Revised{\mbox{\Cref{fig:reference_frames}} shows these frames together with the pose
error between the current and the desired end-effector frame.}

  \label{fig:reference_frames}
\end{figure}

\section{Rigid-Body Transformations}
\label{sec:rigid_transformations}

\Revised{The positions and orientations of the frames introduced in
\mbox{\Cref{sec:reference_frames}} are related through rotation matrices and
homogeneous transformations.}


\subsection{Rotation Matrices}
\label{sec:rotation_matrices}

\Revised{An orientation is represented by a rotation matrix
\mbox{\(R\in SO(3)\)}, satisfying}

\begin{equation}
R^\top R=I,
\qquad
\det R=1,
\qquad
R^{-1}=R^\top.
\end{equation}

\Revised{The notation \mbox{\({}^bR_a\)} denotes the orientation of frame \mbox{\(\{a\}\)}
expressed in frame \mbox{\(\{b\}\)}. Its columns contain the axes of frame
\mbox{\(\{a\}\)} expressed in frame \mbox{\(\{b\}\)}. A vector is transformed from frame
\mbox{\(\{a\}\)} to frame \mbox{\(\{b\}\)} according to}

\begin{equation}
{}^bv = {}^bR_a\,{}^av.
\label{eq:rotation_transformation}
\end{equation}

\Revised{The elementary right-handed rotations used later to define the commanded tool
tilt are}

\begin{equation}
% Tightened so the whole row plus the equation tag fits the text width: at the
% default column spacing the tag is forced onto a line of its own.
\setlength{\arraycolsep}{1pt}
R_x(\alpha)=
\begin{bmatrix}
1&0&0\\
0&\cos\alpha&-\sin\alpha\\
0&\sin\alpha&\cos\alpha
\end{bmatrix},\;
R_y(\beta)=
\begin{bmatrix}
\cos\beta&0&\sin\beta\\
0&1&0\\
-\sin\beta&0&\cos\beta
\end{bmatrix},\;
R_z(\gamma)=
\begin{bmatrix}
\cos\gamma&-\sin\gamma&0\\
\sin\gamma&\cos\gamma&0\\
0&0&1
\end{bmatrix}.
\label{eq:elementary_rotations}
\end{equation}

\subsection{Homogeneous Transformations}
\label{sec:homogeneous_transformations}

\Revised{A position is represented by a vector \mbox{\(p\in\mathbb{R}^3\)}. Rotation and
translation are combined in a homogeneous transformation
\mbox{\cite{Craig2005,Siciliano2009}}:}

\begin{equation}
{}^bT_a=
\begin{bmatrix}
{}^bR_a & {}^bp_a\\
0^\top & 1
\end{bmatrix},
\label{eq:homogeneous_transformation}
\end{equation}

\Revised{where \mbox{\({}^bp_a\)} is the position of the origin of frame \mbox{\(\{a\}\)} expressed
in frame \mbox{\(\{b\}\)}. A point expressed in frame \mbox{\(\{a\}\)} is transformed to
frame \mbox{\(\{b\}\)} according to}

\begin{equation}
{}^bp
=
{}^bR_a\,{}^ap+{}^bp_a.
\label{eq:point_transformation}
\end{equation}

\Revised{Successive transformations are composed by matrix multiplication:}

\begin{equation}
{}^cT_a
=
{}^cT_b\,{}^bT_a.
\label{eq:transformation_composition}
\end{equation}




\subsection{End-Effector Pose}
\label{sec:position_orientation}

\Revised{The controlled pose consists of the end-effector position}

\begin{equation}
p_{\mathrm{EE}}
=
\begin{bmatrix}
p_{x,\mathrm{EE}} &
p_{y,\mathrm{EE}} &
p_{z,\mathrm{EE}}
\end{bmatrix}^{\top}
\in\mathbb{R}^3
\label{eq:end_effector_position}
\end{equation}

\Revised{and the orientation \mbox{\(R_{\mathrm{EE}}\in SO(3)\)}. Both are expressed in the
base frame and combined in the homogeneous transformation}

\begin{equation}
T_{\mathrm{EE}}
=
\begin{bmatrix}
R_{\mathrm{EE}} & p_{\mathrm{EE}}\\
0^\top & 1
\end{bmatrix}.
\label{eq:end_effector_pose}
\end{equation}

\Revised{In the implementation, \mbox{\(T_{\mathrm{EE}}\)} is obtained directly from the
reported robot state, and the orientation is not converted to an Euler-angle
representation.}


\subsection{Cartesian Pose Error}
\label{subsec:cartesian_pose_error}

\Revised{Let the desired end-effector pose be}

\begin{equation}
T_d
=
\begin{bmatrix}
R_d & p_d\\
0^\top & 1
\end{bmatrix},
\label{eq:desired_transformation}
\end{equation}

\Revised{where \mbox{\(p_d\)} and \mbox{\(R_d\)} are the desired TCP position and orientation,
respectively. The current end-effector pose is denoted by
\mbox{\(T_{\mathrm{EE}}\)}.}

\Revised{The relative transformation from the current end-effector frame to the desired
frame is}

\begin{equation}
T_{\mathrm{err}}
=
T_{\mathrm{EE}}^{-1}T_d
=
\begin{bmatrix}
R_{\mathrm{EE}}^\top R_d &
R_{\mathrm{EE}}^\top(p_d-p_{\mathrm{EE}})\\
0^\top & 1
\end{bmatrix}.
\label{eq:expanded_transformation_error}
\end{equation}

\Revised{When the current and desired poses coincide,
\mbox{\(T_{\mathrm{err}}=I_4\)}. The rotational and translational blocks of
\mbox{\cref{eq:expanded_transformation_error}} therefore both represent zero error in
this case.}

\paragraph{Position error.}

\Revised{The translational block of \mbox{\cref{eq:expanded_transformation_error}} expresses
the position error in the current end-effector frame. The Cartesian impedance
law uses the equivalent restoring error expressed in the base frame:}

\begin{equation}
e_p
=
p_d-p_{\mathrm{EE}}
\in\mathbb{R}^3.
\label{eq:position_control_error}
\end{equation}

\Revised{A positive position error therefore points from the current TCP position toward
the desired TCP position.}

\paragraph{Orientation error.}

\Revised{The rotational block of \mbox{\cref{eq:expanded_transformation_error}} is}

\begin{equation}
\Delta R
=
R_{\mathrm{EE}}^\top R_d.
\label{eq:relative_rotation_error}
\end{equation}

\Revised{It represents the rotation required to carry the current end-effector
orientation to the desired orientation. If both orientations coincide, then
\mbox{\(\Delta R=I\)}.}

\Revised{For use in the impedance controller, \mbox{\(\Delta R\)} is represented by a rotation
angle \mbox{\(\phi\)} and a unit rotation axis \mbox{\(u\in\mathbb{R}^3\)}
\mbox{\cite{Siciliano2009,Murray1994}}. The angle follows from}

\begin{equation}
\operatorname{tr}(\Delta R)
=
1+2\cos\phi,
\label{eq:trace_angle_relation}
\end{equation}

\Revised{and is therefore}

\begin{equation}
\phi
=
\arccos\!\left(
\frac{\operatorname{tr}(\Delta R)-1}{2}
\right),
\qquad
\phi\in[0,\pi].
\label{eq:angle_from_trace}
\end{equation}

\Revised{Rodrigues' formula expresses the relative rotation as}

\begin{equation}
\Delta R
=
I
+\sin\phi\,[u]_\times
+(1-\cos\phi)[u]_\times^2,
\label{eq:rodrigues_formula}
\end{equation}

\Revised{where}

\begin{equation}
[u]_\times
=
\begin{bmatrix}
0 & -u_z & u_y\\
u_z & 0 & -u_x\\
-u_y & u_x & 0
\end{bmatrix},
\qquad
[u]_\times^\top=-[u]_\times.
\label{eq:skew_symmetric_axis_matrix}
\end{equation}

\Revised{Subtracting the transpose of \mbox{\(\Delta R\)} isolates its skew-symmetric part:}

\begin{equation}
\frac{1}{2}
\left(
\Delta R-\Delta R^\top
\right)
=
\sin\phi\,[u]_\times.
\label{eq:skew_part_rotation_matrix}
\end{equation}

\Revised{For \mbox{\(\sin\phi\neq0\)}, the rotation axis is obtained from}

\begin{equation}
u
=
\frac{1}{2\sin\phi}
\begin{bmatrix}
\Delta R_{32}-\Delta R_{23}\\
\Delta R_{13}-\Delta R_{31}\\
\Delta R_{21}-\Delta R_{12}
\end{bmatrix}.
\label{eq:rotation_axis_from_matrix}
\end{equation}

\Revised{Multiplying the unit axis by the angle gives the orientation-error vector in
the current end-effector frame. It is then rotated into the base frame:}

\begin{equation}
e_{R,\mathrm{EE}}
=
\phi u,
\qquad
e_R
=
R_{\mathrm{EE}}e_{R,\mathrm{EE}}
\in\mathbb{R}^3.
\label{eq:rotation_error_base_frame}
\end{equation}

\Revised{The direction of \mbox{\(e_R\)} defines the axis of the required correction, while
its magnitude is the angular error in radians. When the current and desired
orientations coincide, \mbox{\(\phi=0\)} and \mbox{\(e_R=0\)}.}

\Revised{The complete Cartesian error used by the impedance controller is}

\begin{equation}
e
=
\begin{bmatrix}
e_p\\
e_R
\end{bmatrix}
\in\mathbb{R}^6.
\label{eq:cartesian_error_vector}
\end{equation}


\section{Kinematics of a 7-DOF Manipulator}
\label{sec:kinematics}

\Revised{Having defined the end-effector pose as the controlled variable, its relation
to the joint configuration is introduced next.}

\subsection{Forward Kinematics}
\label{subsec:forward_kinematics}

\Revised{Forward kinematics maps the joint configuration}

\begin{equation}
q =
\begin{bmatrix}
q_1 & q_2 & \cdots & q_n
\end{bmatrix}^{\top}
\in\mathbb{R}^n
\label{eq:joint_configuration_vector}
\end{equation}

\Revised{to the end-effector pose. Using the joint frames introduced in
\mbox{\Cref{sec:reference_frames}}, the transformation from the base frame
\mbox{\(\{0\}\)} to the final link frame \mbox{\(\{n\}\)} is}

\begin{equation}
{}^0T_n(q)
=
{}^0T_1(q_1)\,
{}^1T_2(q_2)\cdots
{}^{n-1}T_n(q_n).
\label{eq:forward_kinematics_chain}
\end{equation}

\Revised{In the standard Denavit--Hartenberg convention
\mbox{\cite{Denavit1955,Craig2005,Siciliano2009}}, joint \mbox{\(i\)} rotates by \mbox{\(q_i\)}
about the axis \mbox{\(z_{i-1}\)}. The transformation between two consecutive joint
frames is}

\begin{equation}
{}^{i-1}T_i(q_i)
=
R_z(q_i)\,T_z(d_i)\,T_x(a_i)\,R_x(\alpha_i),
\label{eq:dh_elementary_transformations}
\end{equation}

\Revised{where \mbox{\(a_i\)}, \mbox{\(d_i\)}, and \mbox{\(\alpha_i\)} describe the fixed geometry between
the consecutive joint frames, while \mbox{\(q_i\)} is the revolute-joint coordinate.}

\Revised{The complete transformation gives the end-effector pose in the base frame:}

\begin{equation}
{}^0T_n(q)
=
T_{\mathrm{EE}}(q)
=
\begin{bmatrix}
R_{\mathrm{EE}}(q) & p_{\mathrm{EE}}(q)\\
0^\top & 1
\end{bmatrix}.
\label{eq:forward_kinematics_result}
\end{equation}

\Revised{This convention establishes the joint and frame indexing used in the
subsequent Jacobian formulation. In the implemented controller, the
forward-kinematic chain is not evaluated separately; the current end-effector
pose is obtained from the libfranka model interface.}

\subsection{Differential Kinematics}
\label{subsec:differential_kinematics}

\Revised{Differential kinematics relates the joint velocities to the end-effector
twist:}

\begin{equation}
\dot{x}
=
\begin{bmatrix}
\dot{p}_{\mathrm{EE}}\\
\omega_{\mathrm{EE}}
\end{bmatrix}
=
J(q)\dot{q},
\label{eq:differential_kinematics}
\end{equation}

\Revised{where \mbox{\(\dot{p}_{\mathrm{EE}}\in\mathbb{R}^3\)} and
\mbox{\(\omega_{\mathrm{EE}}\in\mathbb{R}^3\)} are the linear and angular velocities
of the end-effector, respectively. The joint-velocity vector is}

\begin{equation}
\dot{q}
=
\begin{bmatrix}
\dot{q}_1 & \dot{q}_2 & \cdots & \dot{q}_n
\end{bmatrix}^{\top}
\in\mathbb{R}^n,
\label{eq:joint_velocity_vector}
\end{equation}

\Revised{and \mbox{\(J(q)\in\mathbb{R}^{6\times n}\)} is the geometric Jacobian. For the
7-\mbox{\abbr{DOF}} Panda, \mbox{\(n=7\)}. The end-effector twist and the Jacobian used in
the controller are expressed in the base frame. This relation provides the
Cartesian velocity required by the damping term of the impedance controller.}


\subsection{Geometric Jacobian}
\label{subsec:geometric_jacobian}

\Revised{The geometric Jacobian is written column-wise as}

\begin{equation}
J(q)
=
\begin{bmatrix}
J_1 & J_2 & \cdots & J_n
\end{bmatrix}
\in\mathbb{R}^{6\times n},
\label{eq:geometric_jacobian}
\end{equation}

\Revised{where \mbox{\(J_i\)} describes the contribution of joint \mbox{\(i\)} to the end-effector
twist. Since all joints of the Panda are revolute, each column is given by}

\begin{equation}
J_i
=
\begin{bmatrix}
z_{i-1}\times
\left(p_{\mathrm{EE}}-p_{i-1}\right)\\
z_{i-1}
\end{bmatrix}
\in\mathbb{R}^6,
\label{eq:jacobian_column}
\end{equation}

\Revised{where \mbox{\(z_{i-1}\)} is the axis of joint \mbox{\(i\)}, and \mbox{\(p_{i-1}\)} is a point on
that axis. Both are expressed in the base frame. The upper block describes the
linear velocity generated at the end-effector, while the lower block describes
the corresponding angular velocity
\mbox{\cite{Siciliano2009,Craig2005}}.}

\Revised{Because the joint axes and their positions depend on the current configuration,
the Jacobian also depends on \mbox{\(q\)}. In the implemented controller,
\mbox{\(J(q)\)} is obtained directly from the libfranka model interface.}


\Revised{The Jacobian describes the local relation between joint velocities and the
end-effector twist. For a robot with \mbox{\(n\)} joints it is written column-wise,}
\begin{equation}
\begin{bmatrix}\dot p_{\mathrm{EE}}\\\omega_{\mathrm{EE}}\end{bmatrix}
=
J(q)\dot q,
\qquad
J(q)=\begin{bmatrix}J_1&J_2&\cdots&J_n\end{bmatrix}
\in\mathbb{R}^{6\times n},
\label{eq:jacobian_velocity_relation}
\end{equation}
\Revised{where each column \mbox{\(J_i\)} is the contribution of joint \mbox{\(i\)}. The Panda has
only revolute joints, so with \mbox{\(z_{i-1}\)} the unit vector along the axis of
joint \mbox{\(i\)} and \mbox{\(p_{i-1}\)} a point on that axis, both expressed in the base
frame, the column is}
\begin{equation}
J_i=
\begin{bmatrix}
z_{i-1}\times\left(p_{\mathrm{EE}}-p_{i-1}\right)\\
z_{i-1}
\end{bmatrix}
\in\mathbb{R}^6 .
\label{eq:jacobian_column_revolute}
\end{equation}
\Revised{This standard form follows from the instantaneous velocity of a point under a
revolute motion \mbox{\cite{Siciliano2009,Craig2005}}. The vector
\mbox{\(p_{\mathrm{EE}}-p_{i-1}\)} is the lever arm from the joint axis to the
end-effector, so the cross product
\mbox{\(z_{i-1}\times(p_{\mathrm{EE}}-p_{i-1})\)} is the linear velocity that rotation
of joint \mbox{\(i\)} produces at the end-effector, while the lower block \mbox{\(z_{i-1}\)}
is the corresponding angular velocity. Each column therefore stacks a linear
contribution above an angular one.}

\Revised{Because \mbox{\(z_{i-1}\)}, \mbox{\(p_{i-1}\)} and \mbox{\(p_{\mathrm{EE}}\)} all depend on the
current joint configuration, the Jacobian is configuration dependent, and the
influence of each joint on the end-effector motion changes as the robot moves.}

\section{Cartesian Impedance Control}
\label{sec:cartesian_impedance_control}

\Revised{The geometric Jacobian is also used through its transpose to map a Cartesian
wrench to joint torques \mbox{\cite{Khatib1987}}. Before introducing this mapping, the
desired Cartesian interaction behaviour is defined.}

\Revised{In this thesis, Cartesian impedance control is formulated as a
spring--damper relation between the end-effector motion and the commanded
wrench \mbox{\cite{Hogan1985,Ott2008,Mayr2024CartesianImpedance}}. The Cartesian pose
error \mbox{\(e\)}, defined in \mbox{\Cref{subsec:cartesian_pose_error}}, produces a
restoring wrench through the stiffness matrix \mbox{\(K\)}, while the Cartesian
velocity produces an opposing wrench through the damping matrix \mbox{\(D\)}. Their
entries determine the translational and rotational compliance in the selected
task directions \mbox{\cite{Pollayil2023ImpedancePlanning}}.}






\subsection{Impedance Wrench}
\label{subsec:impedance_wrench}

\Revised{The impedance wrench is evaluated in the robot base frame. The corresponding
translational stiffness and damping matrices are
\mbox{\(K_{p,0},D_{p,0}\in\mathbb{R}^{3\times3}\)}, while the rotational matrices are
\mbox{\(K_{R,0},D_{R,0}\in\mathbb{R}^{3\times3}\)}. These matrices are not generally
diagonal in the base frame because the direction-dependent gains are defined
relative to the task frame. Their base-frame representation therefore changes
with the task-frame orientation, even when the local gain values remain
unchanged.}

\Revised{The commanded Cartesian force is given by the translational spring--damper law
\mbox{\cite{Hogan1985,Mayr2024CartesianImpedance}}}

\begin{equation}
f
=
K_{p,0}e_p
+
D_{p,0}
\left(
\dot{p}_d-\dot{p}_{\mathrm{EE}}
\right)
\in\mathbb{R}^3,
\label{eq:positional_impedance_force}
\end{equation}

\Revised{where \mbox{\(\dot{p}_d\)} and \mbox{\(\dot{p}_{\mathrm{EE}}\)} are the desired and measured
linear velocities expressed in the base frame. For a fixed position setpoint,
\mbox{\(\dot{p}_d=\mathbf{0}\)}, and the damping term becomes
\mbox{\(-D_{p,0}\dot{p}_{\mathrm{EE}}\)}.}

\Revised{The commanded Cartesian moment is}

\begin{equation}
m
=
K_{R,0}e_R
-
D_{R,0}\omega_{\mathrm{EE}}
\in\mathbb{R}^3,
\label{eq:rotational_impedance_moment}
\end{equation}

\Revised{where \mbox{\(\omega_{\mathrm{EE}}\)} is the measured end-effector angular velocity
expressed in the base frame. Since no angular reference velocity is commanded,
the rotational damping acts directly against
\mbox{\(\omega_{\mathrm{EE}}\)}.}

\Revised{The force and moment are combined into the Cartesian wrench}

\begin{equation}
F
=
\begin{bmatrix}
f\\
m
\end{bmatrix}
=
\begin{bmatrix}
K_{p,0}e_p
+
D_{p,0}\left(\dot{p}_d-\dot{p}_{\mathrm{EE}}\right)
\\[1mm]
K_{R,0}e_R
-
D_{R,0}\omega_{\mathrm{EE}}
\end{bmatrix}
\in\mathbb{R}^6.
\label{eq:cartesian_force_moment_vector}
\end{equation}

\Revised{The wrench is ordered as force followed by moment, and all its components are
expressed in the base frame. It is mapped to joint torques through the
Jacobian transpose in the following subsection.}

\subsection{Joint-Torque Control Law}
\label{sec:control_law}

\Revised{By the principle of virtual work, the commanded Cartesian wrench is mapped to
joint torques through the transpose of the geometric Jacobian
\mbox{\cite{Khatib1987}}:}

\begin{equation}
\tau_{\mathrm{cart}}
=
J^\top(q)F,
\label{eq:joint_torque_mapping}
\end{equation}

\Revised{where \mbox{\(\tau_{\mathrm{cart}}\in\mathbb{R}^n\)} is the Cartesian impedance
torque contribution and
\mbox{\(J^\top(q)\in\mathbb{R}^{n\times6}\)}. Substituting
\mbox{\cref{eq:cartesian_force_moment_vector}} gives}

\begin{equation}
\tau_{\mathrm{cart}}
=
J^\top(q)
\begin{bmatrix}
K_{p,0}e_p
+
D_{p,0}\left(\dot{p}_d-\dot{p}_{\mathrm{EE}}\right)
\\[1mm]
K_{R,0}e_R
-
D_{R,0}\omega_{\mathrm{EE}}
\end{bmatrix}.
\label{eq:cartesian_impedance_control_law}
\end{equation}

\Revised{The gain matrices in this expression are the base-frame matrices obtained from
the selected task-frame gains. A change in the surface-frame orientation
therefore changes the Cartesian wrench and the resulting joint-torque
contribution, even when the local gain values remain unchanged.}

\subsection{Coriolis and Gravity Compensation in libfranka}
\label{subsec:coriolis_gravity_compensation}

\Revised{In a general model-compensated torque controller, the Cartesian impedance
torque is combined with gravity and Coriolis/centrifugal compensation:}

\begin{equation}
\tau_{\mathrm{cmd}}
=
\tau_{\mathrm{cart}}
+
\tau_g(q)
+
\tau_c(q,\dot{q}),
\label{eq:commanded_torque}
\end{equation}

\Revised{where \mbox{\(\tau_g(q)\)} is the gravity torque and
\mbox{\(\tau_c(q,\dot{q})\)} is the Coriolis and centrifugal torque.}

\Revised{These terms are handled differently by the Franka Control Interface. The
Coriolis vector provided by the libfranka model is added explicitly to the
command because it is not compensated internally. A separate gravity term is
not added. In external joint-torque mode, the robot compensates gravity and
motor friction internally, and the commanded torque is interpreted as excluding
gravity \mbox{\cite{FrankaFCI2023}}. Adding \mbox{\(\tau_g(q)\)} would therefore compensate
gravity twice.}

\Revised{The implemented torque command is consequently}

\begin{equation}
\tau_{\mathrm{cmd,impl}}
=
\tau_{\mathrm{cart}}
+
\tau_c(q,\dot{q})
=
J^\top(q)F
+
\tau_c(q,\dot{q}),
\label{eq:commanded_torque_libfranka}
\end{equation}

\Revised{which follows the official libfranka Cartesian impedance example
\mbox{\cite{libfrankaCartesianExample}}. The null-space torque introduced later is
added to this expression to obtain the complete commanded torque in
\mbox{\cref{eq:nullspace_complete_torque_command}}.}
\section{Surface Task Frame}
\label{subsec:surface_task_frame}
\Revised{The contact experiment uses a task frame tied to the surface geometry. This
frame is defined before the stiffness and damping matrices are introduced, so
that the local gain entries can later be interpreted as normal and tangential
values instead of as abstract base-frame directions.}

\Revised{Let \mbox{\(n_s\in\mathbb{R}^3\)} with \mbox{\(\|n_s\|=1\)} be the unit surface normal,
expressed in the robot base frame. To complete the frame, choose an auxiliary
direction \mbox{\(a_s\in\mathbb{R}^3\)} that is not parallel to \mbox{\(n_s\)}. This auxiliary
direction is only used to define a repeatable tangent direction on the surface.
The first tangent vector is obtained by projecting \mbox{\(a_s\)} onto the plane
orthogonal to \mbox{\(n_s\)}, and the second tangent vector is then obtained by a cross
product. With \mbox{\(\tilde{t}_1\)} denoting the unnormalised projected tangent, the
construction is a Gram--Schmidt orthonormalization step applied to the surface
normal and the auxiliary direction \mbox{\cite{Siciliano2009,Craig2005}}:}

\begin{equation}
\begin{aligned}
\tilde{t}_1
&=
a_s - n_s\left(n_s^\top a_s\right),\\
t_1
&=
\frac{\tilde{t}_1}{\|\tilde{t}_1\|},\\
t_2
&=
n_s\times t_1 .
\end{aligned}
\label{eq:surface_frame_tangents}
\end{equation}

\Revised{Here, \mbox{\(t_1\)} and \mbox{\(t_2\)} are unit tangent vectors lying in the surface plane.
The construction is valid as long as \mbox{\(\|\tilde{t}_1\|\neq0\)}, which means that
the auxiliary direction \mbox{\(a_s\)} must not be parallel to the normal \mbox{\(n_s\)}. To
match the implemented controller and its parameter order, the surface task
frame collects the two tangents first and the normal last,}
\begin{equation}
R_{\mathrm{surface}}
=
\begin{bmatrix}
t_1 & t_2 & n_s
\end{bmatrix}.
\label{eq:surface_task_frame_matrix}
\end{equation}

\Revised{The column order is fixed: every surface-frame three-vector in the
implementation uses
\mbox{\([\,\text{tangent 1},\text{tangent 2},\text{normal}\,]^\top\)}.
In the gain representation below, the contact task uses this frame by setting
\mbox{\(R_{\mathrm{task}}=R_{\mathrm{surface}}\)}. This makes stiffness values such as
\mbox{\(K_{t_1}\)}, \mbox{\(K_{t_2}\)}, and \mbox{\(K_n\)} local surface-frame quantities in that
same order.}


\section{Stiffness and Damping Representation}
\label{sec:stiffness_damping_representation}

\Revised{Direction-dependent stiffness and damping gains are most conveniently defined
in a task frame whose axes coincide with the desired principal compliance
directions. The corresponding matrices are diagonal in that frame but are not
generally diagonal when expressed in the robot base frame.}

\Revised{Let \mbox{\(R_{\mathrm{task}}\in SO(3)\)} denote the orientation of the task frame
relative to the base frame. Its columns contain the task-frame axes expressed
in the base frame. The translational and rotational gain matrices used in the
impedance wrench are obtained from}

\begin{equation}
\begin{aligned}
K_{p,0}
&=
R_{\mathrm{task}}
K_{p,\mathrm{task}}
R_{\mathrm{task}}^\top,
&
D_{p,0}
&=
R_{\mathrm{task}}
D_{p,\mathrm{task}}
R_{\mathrm{task}}^\top,
\\
K_{R,0}
&=
R_{\mathrm{task}}
K_{R,\mathrm{task}}
R_{\mathrm{task}}^\top,
&
D_{R,0}
&=
R_{\mathrm{task}}
D_{R,\mathrm{task}}
R_{\mathrm{task}}^\top.
\end{aligned}
\label{eq:general_task_frame_gain_transform}
\end{equation}

\Revised{For an impedance without translation--rotation coupling in the task frame, the
full stiffness and damping matrices are}

\begin{equation}
K_{\mathrm{task}}
=
\begin{bmatrix}
K_{p,\mathrm{task}} & 0\\
0 & K_{R,\mathrm{task}}
\end{bmatrix},
\qquad
D_{\mathrm{task}}
=
\begin{bmatrix}
D_{p,\mathrm{task}} & 0\\
0 & D_{R,\mathrm{task}}
\end{bmatrix}.
\label{eq:full_task_frame_gain_blocks}
\end{equation}

\Revised{Defining}

\begin{equation}
T_R
=
\operatorname{diag}
\left(
R_{\mathrm{task}},
R_{\mathrm{task}}
\right),
\end{equation}

\Revised{their base-frame representations are}

\begin{equation}
K_0
=
T_R K_{\mathrm{task}}T_R^\top,
\qquad
D_0
=
T_R D_{\mathrm{task}}T_R^\top.
\label{eq:full_task_frame_gain_transform}
\end{equation}

\Revised{For the contact task, the task frame is the surface frame, whose orientation is}

\begin{equation}
R_{\mathrm{task}}
=
R_{\mathrm{surface}}
=
\begin{bmatrix}
t_1 & t_2 & n_s
\end{bmatrix}.
\end{equation}

\Revised{For example, the local translational stiffness is defined as}

\begin{equation}
K_{p,\mathrm{task}}
=
\operatorname{diag}
\left(
K_{t_1},
K_{t_2},
K_n
\right),
\end{equation}

\Revised{and its base-frame representation is}

\begin{equation}
K_{p,0}
=
R_{\mathrm{surface}}\,
\operatorname{diag}
\left(
K_{t_1},
K_{t_2},
K_n
\right)
R_{\mathrm{surface}}^\top.
\label{eq:spatial_gain_matrix}
\end{equation}

\Revised{The rotational stiffness follows the same axis order,}

\begin{equation}
K_{R,\mathrm{task}}
=
\operatorname{diag}
\left(
K_{r,t_1},
K_{r,t_2},
K_{r,n}
\right),
\end{equation}

\Revised{and the damping matrices are constructed analogously. A change in the
configured surface orientation therefore changes the corresponding base-frame
gain matrices even when the local gain values remain unchanged.}

\Revised{During approach, contact set-up, and grinding, the reported contact runs
used this surface-frame construction for both translational and rotational gain
blocks. For set-up and grinding, the translational entries
\mbox{\([2000,2000,800]\,\mathrm{N/m}\)} were assigned to
\mbox{\([t_1,t_2,n_s]\)} before transformation to the base frame. The active gain
matrices were based on the configured surface frame and were not rotated with
the current end-effector orientation \mbox{\(R_{\mathrm{EE}}\)}.}


\section{Centre of Compliance}
\label{sec:coupled_pole_stiffness}

\Revised{The task-frame transformation of
\mbox{\cref{eq:full_task_frame_gain_transform}} rotates the principal stiffness
directions while leaving the impedance reference point at the end-effector
origin, identified here with the tool centre point (TCP). Independently of this
orientation choice, the virtual spring can be centred at a different point.
Shifting the impedance reference from the TCP to a centre of compliance
\mbox{\(p_c\)} transforms a block-diagonal impedance at \mbox{\(p_c\)} into a coupled
impedance at the TCP
\mbox{\cite{Murray1994,Fasse1997SpatialImpedance,Stramigioli2000IPC,LynchPark2017}}.
This construction is applied during contact set-up to generate
translation--rotation coupling that supports tool alignment.}

\Revised{\mbox{\Cref{fig:compliance_lever_moment}} shows this geometry for the two signs
of the tangential lever.}

\begin{figure}[H]
  \centering
  \input{figures/compliance_lever_moment.tex}
  \caption{Moment produced by the compliance-centre lever at both lever signs.}
  \label{fig:compliance_lever_moment}
\end{figure}

\Revised{For a dual-arm assembly task, Suomalainen \mbox{et al.\ }experimentally
showed that compliance-centre placement and compliant-axis selection affect the
alignment response \mbox{\citep{Suomalainen2019DualArmCompliance}}. This task-level
result motivates the placement study in this thesis. The point-shift adjoint
below remains derived from the spatial-transformation formulation.}

\Revised{The lever from the centre of compliance to the TCP, expressed in the base
frame, is}

\begin{equation}
r_c
=
p_{\mathrm{TCP}}-p_c
\in\R{3},
\qquad
p_{\mathrm{TCP}}=p_{\mathrm{EE}}.
\label{eq:pole_to_tcp_lever}
\end{equation}

\Revised{Let \mbox{\([r_c]_\times\)} denote the skew-symmetric matrix satisfying
\mbox{\([r_c]_\times a=r_c\times a\)}. Under the small-angle approximation, the
position error at the centre of compliance is}

\begin{equation}
e_{p,c}
=
e_p+[r_c]_\times e_R.
\label{eq:pole_small_motion_error}
\end{equation}

\Revised{The positive sign follows from the selected lever convention. The vector from
the TCP to the centre of compliance is \mbox{\(-r_c\)}, and therefore}

\[
e_R\times(-r_c)
=
[r_c]_\times e_R.
\]

\Revised{The translational and rotational errors at the centre of compliance can thus be
written as}

\begin{equation}
e_c
=
\begin{bmatrix}
e_{p,c}\\[2pt]
e_R
\end{bmatrix}
=
\mathrm{Ad}(r_c)
\begin{bmatrix}
e_p\\[2pt]
e_R
\end{bmatrix},
\qquad
\mathrm{Ad}(r_c)
=
\begin{bmatrix}
\mI_3 & [r_c]_\times\\[2pt]
0 & \mI_3
\end{bmatrix}.
\label{eq:offset_adjoint}
\end{equation}

\Revised{The point-shift adjoint \mbox{\(\mathrm{Ad}(r_c)\)} changes the reference point of the
impedance. By contrast,
\mbox{\(T_R=\operatorname{diag}(R_{\mathrm{task}},R_{\mathrm{task}})\)} in
\mbox{\cref{eq:full_task_frame_gain_transform}} changes its principal directions.}

\Revised{After any required task-frame rotation, let the decoupled stiffness and damping
matrices defined at the centre of compliance be}

\begin{equation}
K_c
=
\begin{bmatrix}
K_p & 0\\
0 & K_R
\end{bmatrix},
\qquad
D_c
=
\begin{bmatrix}
D_p & 0\\
0 & D_R
\end{bmatrix}.
\label{eq:pole_block_gains}
\end{equation}

\Revised{All blocks in \mbox{\cref{eq:pole_block_gains}} are expressed in the base-frame
orientation. The wrench at the centre of compliance is}

\[
w_c=K_ce_c.
\]

\Revised{Transforming this power-conjugate wrench to the TCP gives}

\[
w_{\mathrm{TCP}}
=
\mathrm{Ad}(r_c)^\top w_c.
\]

\Revised{The resulting TCP stiffness is therefore}

\begin{equation}
K_{\mathrm{TCP}}
=
\mathrm{Ad}(r_c)^\top
K_c
\mathrm{Ad}(r_c)
=
\begin{bmatrix}
K_p
&
K_p[r_c]_\times
\\[4pt]
[r_c]_\times^\top K_p
&
K_R+[r_c]_\times^\top K_p[r_c]_\times
\end{bmatrix}.
\label{eq:coupled_tcp_stiffness}
\end{equation}

\Revised{The same point shift is applied to the damping matrix:}

\begin{equation}
D_{\mathrm{TCP}}
=
\mathrm{Ad}(r_c)^\top
D_c
\mathrm{Ad}(r_c)
=
\begin{bmatrix}
D_p
&
D_p[r_c]_\times
\\[4pt]
[r_c]_\times^\top D_p
&
D_R+[r_c]_\times^\top D_p[r_c]_\times
\end{bmatrix}.
\label{eq:coupled_tcp_damping}
\end{equation}

\Revised{The off-diagonal blocks introduce translation--rotation coupling. A
translational error can therefore generate a restoring moment, while a
rotational error can generate a restoring force. This coupling is absent from
the block-diagonal impedance defined directly at the TCP.}

\Revised{The sign of the translation-to-moment coupling can be checked by setting
\mbox{\(e_R=0\)}. With}

\[
f=K_pe_p,
\]

\Revised{the corresponding elastic moment at the TCP is}

\begin{equation}
m_c
=
[r_c]_\times^\top f
=
-r_c\times f.
\label{eq:pole_translation_moment_sign_check}
\end{equation}

\Revised{Reversing the direction of \mbox{\(r_c\)} therefore reverses the generated coupling
moment. This relation provides a direct sign check for the selected centre of
compliance and pressing direction.}

\Revised{The coupling vanishes for \mbox{\(r_c=0\)}. Conversely, if \mbox{\(K_p\)} is nonsingular,
zero coupling implies \mbox{\(r_c=0\)}. Since \mbox{\(\mathrm{Ad}(r_c)\)} is invertible, the
congruence transformation also preserves definiteness:}

\begin{equation}
K_c\succeq0
\ \Longrightarrow\
K_{\mathrm{TCP}}\succeq0,
\qquad
K_c\succ0
\ \Longrightarrow\
K_{\mathrm{TCP}}\succ0.
\label{eq:pole_gain_definiteness}
\end{equation}

\Revised{The same statements apply to \mbox{\(D_c\)} and \mbox{\(D_{\mathrm{TCP}}\)}. Negative
off-diagonal entries therefore represent coupling terms rather than negative
stiffness directions.}

\Revised{The congruence transformation preserves the symmetry and definiteness of
the source gain matrices. Closed-loop robot--controller--contact behaviour with
phase switching was assessed experimentally under the operating conditions
reported in \mbox{\Cref{ch:methodology}}.}

\Revised{This formulation is a local small-motion model in which \mbox{\(r_c\)} is held fixed
while the wrench is evaluated. In the implemented set-up mode, the selected
lever remains constant, and the mapped gain matrices are therefore constant
during the corresponding motion segment. If \mbox{\(p_c\)} or \mbox{\(r_c\)} were updated
continuously, the gain matrices would become configuration dependent.}

\Revised{The selected point \mbox{\(p_c\)} is consequently a local linearised centre of
compliance rather than a measured physical pivot. It need not coincide with
the physical contact point \mbox{\(p_C\)}. The observed motion also depends on the
finite rotational stiffness, the lever direction, the environmental
constraint, friction, and the additional active controller terms introduced
later.}

\section{Null-Space Torque}
\label{sec:nullspace_svd}

\Revised{The Franka Emika Panda has seven joints, while the controlled end-effector task
contains six components: three translational and three rotational. The robot is
therefore kinematically redundant. The Cartesian impedance behaviour at the
end-effector is the primary objective, and the remaining joint-space freedom can
be used for a secondary objective.}

\Revised{In this thesis, the secondary joint-space action consists of a null-space
damping contribution and an active singular-value conditioning contribution.
Both are projected using the joint-torque null-space projector \mbox{\(N_\tau\)}.
The required null-space direction, pseudoinverse, and projector are introduced
before the two torque contributions are defined.}


\subsection{SVD-Based Null-Space Characterisation}
\label{subsec:svd_nullspace_characterisation}

\Revised{The geometric Jacobian relates the current joint velocity \mbox{\(\dot q\)} to the
end-effector velocity \mbox{\(\dot x\)}:}

\begin{equation}
\dot{x}
=
J(q)\dot q,
\qquad
J(q)\in\mathbb{R}^{6\times7},
\label{eq:nullspace_jacobian_relation}
\end{equation}

\Revised{where}

\begin{equation}
\dot{x}
=
\begin{bmatrix}
\dot p_{\mathrm{EE}}\\
\omega_{\mathrm{EE}}
\end{bmatrix}
\in\mathbb{R}^6
\label{eq:nullspace_end_effector_velocity}
\end{equation}

\Revised{contains the three linear and three angular velocity components of the
end-effector.}

\Revised{A joint velocity \mbox{\(\dot q\)} belongs to the kinematic null space when}

\begin{equation}
J(q)\dot q
=
0.
\label{eq:jacobian_nullspace_condition}
\end{equation}

\Revised{Such a joint velocity changes the robot configuration while preserving the
end-effector position and orientation to first order. The Jacobian is the
first-order term in the expansion of the forward kinematics about the current
configuration, so a finite displacement along the null space leaves a pose error
of second order in its magnitude. When \mbox{\(J(q)\)} has full
row rank, the null space of the \mbox{\(6\times7\)} Jacobian is one-dimensional.}

\Revised{The singular value decomposition (\mbox{\abbr{SVD}}) is applied to the Jacobian at
the current joint configuration \mbox{\(q\)}
\mbox{\cite{Chiaverini1994DampedLeastSquares,Dietrich2015NullspaceSurvey}}:}

\begin{equation}
J(q)
=
U_J\Sigma_JV_J^\top,
\qquad
U_J\in\mathbb{R}^{6\times6},
\quad
\Sigma_J\in\mathbb{R}^{6\times7},
\quad
V_J\in\mathbb{R}^{7\times7}.
\label{eq:jacobian_svd}
\end{equation}

\Revised{The Jacobian \mbox{\(J(q)\)} is the input to the decomposition. The \mbox{\abbr{SVD}}
returns the matrices \mbox{\(U_J\)}, \mbox{\(\Sigma_J\)}, and \mbox{\(V_J\)}, with the singular
values contained in \mbox{\(\Sigma_J\)}.}

\Revised{The columns of \mbox{\(V_J\)} form an orthonormal basis of the
seven-dimensional joint-velocity space. The joint velocity \mbox{\(\dot q\)} can
therefore be expressed as}

\begin{equation}
\dot q
=
V_Jc
=
\sum_{i=1}^{7}c_i v_i,
\label{eq:joint_velocity_svd_basis}
\end{equation}

\Revised{where \mbox{\(v_i\)} is the \mbox{\(i\)}-th column of \mbox{\(V_J\)}, and
\mbox{\(c\in\mathbb{R}^7\)} contains the components of \mbox{\(\dot q\)} along these basis
directions.}

\Revised{Similarly, the columns of \mbox{\(U_J\)} form an orthonormal basis of the
six-dimensional end-effector-velocity space. Substituting
\mbox{\(\dot q=V_Jc\)} into \mbox{\cref{eq:nullspace_jacobian_relation}} gives}

\begin{equation}
\dot x
=
J(q)V_Jc
=
U_J\Sigma_Jc.
\label{eq:end_effector_velocity_svd_basis}
\end{equation}

\Revised{Thus, \mbox{\(V_J\)} provides the basis directions in which the joint velocity
\mbox{\(\dot q\)} is expressed, while \mbox{\(U_J\)} provides the corresponding basis
directions in which the end-effector velocity \mbox{\(\dot x\)} is expressed. The
matrix \mbox{\(\Sigma_J\)} scales the mapping between these directions.}

\Revised{For the \mbox{\(6\times7\)} Jacobian, the singular-value matrix has the form}

\begin{equation}
\Sigma_J
=
\begin{bmatrix}
\operatorname{diag}
\left(
\sigma_1,\sigma_2,\ldots,\sigma_6
\right)
&
0_{6\times1}
\end{bmatrix},
\label{eq:singular_value_matrix_structure}
\end{equation}

\Revised{where}

\begin{equation}
\sigma_1
\geq
\sigma_2
\geq
\cdots
\geq
\sigma_6
\geq
0.
\label{eq:ordered_singular_values}
\end{equation}

\Revised{For the first six basis directions,}

\begin{equation}
Jv_i
=
\sigma_i u_i,
\qquad
i=1,\ldots,6,
\label{eq:svd_direction_mapping}
\end{equation}

\Revised{where \mbox{\(u_i\)} is the \mbox{\(i\)}-th column of \mbox{\(U_J\)}. A joint velocity in the
direction \mbox{\(v_i\)},}

\begin{equation}
\dot q
=
c_i v_i,
\end{equation}

\Revised{therefore produces the end-effector velocity}

\begin{equation}
\dot x
=
c_i\sigma_i u_i.
\end{equation}

\Revised{The singular value \mbox{\(\sigma_i\)} scales the mapping from the joint-velocity
basis direction \mbox{\(v_i\)} to the corresponding end-effector-velocity basis
direction \mbox{\(u_i\)}.}

\Revised{The seventh column of \mbox{\(V_J\)} corresponds to the zero column of
\mbox{\(\Sigma_J\)} and therefore satisfies}

\begin{equation}
Jv_7
=
0.
\label{eq:nullspace_search_direction}
\end{equation}

\Revised{A joint velocity of the form}

\begin{equation}
\dot q
=
a_7v_7
\end{equation}

\Revised{therefore changes the joint configuration without producing an
end-effector velocity. When the Jacobian has full row rank, \mbox{\(v_7\)} spans its
one-dimensional kinematic null space.}

\Revised{The smallest of the six task-producing singular values is}

\begin{equation}
\sigma_{\min}
=
\sigma_6.
\label{eq:min_singular_value}
\end{equation}

\Revised{The quantities \mbox{\(v_7\)} and \mbox{\(\sigma_6\)} have different purposes. The vector
\mbox{\(v_7\)} represents the redundant joint-motion direction, whereas
\mbox{\(\sigma_6\)} describes the weakest of the six mappings from joint velocity to
end-effector velocity. As \mbox{\(\sigma_6\)} approaches zero, one end-effector
velocity direction becomes increasingly difficult to produce, indicating an
approach to a loss of Jacobian row rank. In this thesis,
\mbox{\(\sigma_{\min}\)} is used to compare the conditioning of nearby joint
configurations and to select the direction of the conditioning torque.}


\subsection{SVD-Based Pseudoinverse}
\label{subsec:svd_pseudoinverse}

\Revised{The Jacobian pseudoinverse maps end-effector-velocity components back to the
corresponding joint-velocity directions. Its calculation uses the singular
directions obtained from the \mbox{\abbr{SVD}}.}

\Revised{Small singular values lead to large reciprocal values and can amplify
numerical errors. A relative singular-value threshold is therefore applied.
Let \mbox{\(\rho\geq0\)} denote the configured relative \mbox{\abbr{SVD}} tolerance. The
corresponding absolute threshold is calculated from the largest singular value
\mbox{\(\sigma_1\)}:}

\begin{equation}
\sigma_{\mathrm{thr}}
=
\rho\sigma_1.
\label{eq:svd_singular_value_threshold}
\end{equation}

\Revised{A singular direction is retained when}

\begin{equation}
\sigma_i
>
\sigma_{\mathrm{thr}}.
\label{eq:svd_retention_condition}
\end{equation}

\Revised{The pseudoinverse is defined as}

\begin{equation}
J^+
=
\sum_{\substack{i=1\\
\sigma_i>\sigma_{\mathrm{thr}}}}^{6}
\frac{1}{\sigma_i}
v_i u_i^\top
\in\mathbb{R}^{7\times6}.
\label{eq:svd_pseudoinverse}
\end{equation}

\Revised{For each retained singular direction, \mbox{\(J^+\)} maps an end-effector-velocity
component along \mbox{\(u_i\)} to the corresponding joint-velocity direction \mbox{\(v_i\)}
and scales it by \mbox{\(1/\sigma_i\)}. Directions at or below
\mbox{\(\sigma_{\mathrm{thr}}\)} receive a reciprocal value of zero.}


\subsection{Null-Space Projectors}
\label{subsec:nullspace_projectors}

\Revised{The pseudoinverse is used to separate a joint-space vector into a
task-related component and a null-space component. For an arbitrary joint
velocity \mbox{\(\dot q\)}, this decomposition is}

\begin{equation}
\dot q
=
\underbrace{J^+J\dot q}_{\text{task-related component}}
+
\underbrace{\left(I_7-J^+J\right)\dot q}_{\text{null-space component}}.
\label{eq:joint_velocity_task_null_decomposition}
\end{equation}

\Revised{The product \mbox{\(J^+J\)} projects \mbox{\(\dot q\)} onto the joint-space directions
that contribute to the end-effector velocity. Since \mbox{\(J\)} includes all three
linear and all three angular velocity components, this task-related component
accounts for the complete controlled end-effector pose at the velocity level.}

\Revised{The complementary joint-velocity projector is}

\begin{equation}
N_q
=
I_7-J^+J
\in\mathbb{R}^{7\times7}.
\label{eq:nullspace_velocity_projector}
\end{equation}

\Revised{Applying \mbox{\(N_q\)} to an arbitrary joint velocity retains its null-space
component:}

\begin{equation}
\dot q_{\mathrm{null}}
=
N_q\dot q.
\label{eq:projected_nullspace_velocity}
\end{equation}

\Revised{When \mbox{\(J\)} has full row rank and all six task-producing singular directions
are retained,}

\begin{equation}
J\dot q_{\mathrm{null}}
=
JN_q\dot q
=
0.
\label{eq:projected_velocity_null_condition}
\end{equation}

\Revised{The projected joint velocity can therefore change the robot configuration
while preserving the end-effector position and orientation to first order.}

\Revised{A corresponding decomposition is used for a secondary joint torque
\mbox{\(\tau_0\)}. The joint-torque projector is the transpose of the velocity
projector:}

\begin{equation}
N_\tau
=
N_q^\top
=
I_7-J^\top J^{+\top}
\in\mathbb{R}^{7\times7}.
\label{eq:nullspace_torque_projector}
\end{equation}

\Revised{The secondary torque can then be written as}

\begin{equation}
\tau_0
=
\underbrace{J^\top J^{+\top}\tau_0}_{\text{task-related component}}
+
\underbrace{N_\tau\tau_0}_{\text{null-space component}}.
\label{eq:joint_torque_task_null_decomposition}
\end{equation}

\Revised{The first component lies in the joint-torque subspace associated with
Cartesian wrenches through \mbox{\(J^\top\)}. The second component is retained as the
secondary null-space torque:}

\begin{equation}
\tau_{\mathrm{null}}
=
N_\tau\tau_0.
\label{eq:general_projected_nullspace_torque}
\end{equation}

\Revised{The transpose relation follows from the mechanical power
\mbox{\(\tau^\top\dot q\)} between joint torques and joint velocities. Thus,
\mbox{\(N_q\)} is applied to joint velocities, while \mbox{\(N_\tau\)} is applied to
secondary joint torques.}

\Revised{For the \mbox{\abbr{SVD}}-based pseudoinverse, \mbox{\(J^+J\)} is an orthogonal and
symmetric projector. Hence,}

\begin{equation}
N_q
=
N_\tau
\label{eq:equal_velocity_torque_projectors}
\end{equation}

\Revised{holds analytically. The separate symbols distinguish the quantity on which the
projector acts. In the implementation, \mbox{\(N_\tau\)} is additionally symmetrised
to reduce floating-point asymmetry.}

\Revised{When \mbox{\(J\)} has full row rank and all six task-producing singular directions
are retained,}

\begin{equation}
N_q
=
N_\tau
=
v_7v_7^\top.
\label{eq:full_rank_nullspace_projector}
\end{equation}

\Revised{In this case, projection retains only the component along the redundant
direction \mbox{\(v_7\)}. When a singular value lies at or below the selected
threshold, its associated right-singular direction is also assigned to the
numerical null space.}


\subsection{Null-Space Damping Torque}
\label{subsec:nullspace_damping_torque}

\Revised{The current joint velocity is separated into task-related and null-space
components according to
\mbox{\cref{eq:joint_velocity_task_null_decomposition}}. The damping contribution
acts on the null-space component \mbox{\(N_q\dot q\)}:}

\begin{equation}
\tau_d
=
-d_{\mathrm{null}}N_\tau\dot q
=
-d_{\mathrm{null}}N_q\dot q,
\label{eq:nullspace_damping_torque}
\end{equation}

\Revised{where \mbox{\(d_{\mathrm{null}}\geq0\)} is the null-space damping coefficient. The
negative sign makes the torque oppose motion along the redundant joint-space
direction and dissipate the associated motion.}


\subsection{Singular-Value Conditioning Torque}
\label{subsec:singular_value_conditioning_torque}

\Revised{The second contribution applies an active torque along the redundant direction
\mbox{\(v_7\)}. Its sign is selected by comparing \mbox{\(\sigma_{\min}\)} at two nearby
joint configurations.}

\Revised{Starting from the current joint configuration \mbox{\(q\)}, the sampled configurations
are}

\begin{equation}
q_+
=
q+\alpha_{\mathrm{probe}}v_7,
\qquad
q_-
=
q-\alpha_{\mathrm{probe}}v_7.
\label{eq:nullspace_candidates}
\end{equation}

\Revised{The Jacobian is recalculated at both configurations. The corresponding smallest
singular values are}

\begin{equation}
\sigma_+
=
\sigma_{\min}\!\left(J(q_+)\right),
\qquad
\sigma_-
=
\sigma_{\min}\!\left(J(q_-)\right).
\label{eq:nullspace_sampled_singular_values}
\end{equation}

\Revised{Their difference is}

\begin{equation}
\Delta_\sigma
=
\sigma_+-\sigma_-.
\label{eq:nullspace_singular_value_difference}
\end{equation}

\Revised{The direction selector is defined as}

\begin{equation}
s
=
\begin{cases}
+1, & \Delta_\sigma>\delta_\sigma,\\
-1, & \Delta_\sigma<-\delta_\sigma,\\
0,  & |\Delta_\sigma|\leq\delta_\sigma,
\end{cases}
\label{eq:nullspace_direction_selection}
\end{equation}

\Revised{where \mbox{\(\delta_\sigma\geq0\)} is the comparison deadband. Within this deadband,
the selector is set to zero and the conditioning contribution is suppressed.}

\Revised{The selected direction is invariant to the sign convention of \mbox{\(v_7\)}
returned by the \mbox{\abbr{SVD}}. Reversing \mbox{\(v_7\)} exchanges \mbox{\(q_+\)} and \mbox{\(q_-\)}
and reverses \mbox{\(s\)}, so the product \mbox{\(sv_7\)} remains unchanged.}

\Revised{The probe angle \mbox{\(\alpha_{\mathrm{probe}}\)} defines the joint-space distance
between the current configuration and the two sampled configurations. The
magnitude of the resulting torque is specified independently by
\mbox{\(k_\sigma\)}.}

\Revised{The vector \mbox{\(v_7\)} represents the instantaneous null-space direction at the
current configuration. For a finite probe angle, \mbox{\(q_+\)} and \mbox{\(q_-\)} provide
local approximations of the nonlinear constant-pose self-motion. The
corresponding preservation of the end-effector pose is therefore a first-order
property of the sampled direction.}

\Revised{The selector uses the sign of \mbox{\(\Delta_\sigma\)} to choose the sampled
direction with the larger value of \mbox{\(\sigma_{\min}\)}. This produces a
two-point directional conditioning strategy based on local comparison rather
than on the magnitude of the sampled variation.}

\Revised{The vector \mbox{\(sv_7\)} defines the selected secondary joint-space direction.
Projecting it with \mbox{\(N_\tau\)} retains its component in the numerical
joint-torque null space:}

\begin{equation}
\tau_\sigma
=
k_\sigma N_\tau\left(sv_7\right),
\label{eq:nullspace_sigma_torque}
\end{equation}

\Revised{where \mbox{\(k_\sigma\geq0\)} specifies the commanded torque magnitude.}

\Revised{When the Jacobian has full row rank and all six task-producing singular
directions are retained, \mbox{\(v_7\)} lies in the one-dimensional null space and}

\begin{equation}
N_\tau v_7
=
v_7.
\label{eq:nullspace_projection_of_v7}
\end{equation}

\Revised{The conditioning contribution therefore biases the robot along the selected
redundant joint direction, while the damping contribution opposes motion
within the projected subspace.}

\subsection{Complete Null-Space Torque}
\label{subsec:complete_nullspace_torque}

\Revised{Combining the damping and conditioning contributions gives}

\begin{equation}
\tau_{\mathrm{null}}
=
\tau_d+\tau_\sigma
=
-d_{\mathrm{null}}N_\tau\dot q
+
k_\sigma N_\tau\left(sv_7\right)
\in\mathbb{R}^7.
\label{eq:nullspace_torque}
\end{equation}

\Revised{When the Jacobian has full row rank, its null space is one-dimensional and is
spanned by \mbox{\(v_7\)}. If the Jacobian loses row rank, the null space becomes
multidimensional and the projector includes the additional numerical-null-space
directions. The damping contribution acts on motion in all these directions
through \mbox{\(N_\tau\dot q\)}. The implemented conditioning term retains its two
probes along \mbox{\(+v_7\)} and \mbox{\(-v_7\)}.}

\Revised{The activation of the damping and conditioning contributions is a controller
configuration choice. The available combinations and the configuration used
in the reported experiments are described in
\mbox{\cref{sec:nullspace_conditioning_implementation}}.}
\section{Complete Joint-Torque Command}
\label{sec:complete_joint_torque_command}

\Revised{The complete joint-torque command combines the Cartesian impedance torque, the
null-space torque, and the Coriolis compensation:}

\begin{equation}
\tau_{\mathrm{cmd}}
=
\tau_{\mathrm{cart}}
+
\tau_{\mathrm{null}}
+
\tau_c(q,\dot q)
=
J^\top(q)F
+
\tau_{\mathrm{null}}
+
\tau_c(q,\dot q).
\label{eq:nullspace_complete_torque_command}
\end{equation}

\Revised{Here, \mbox{\(\tau_{\mathrm{cart}}=J^\top(q)F\)} is the Cartesian impedance torque,
\mbox{\(\tau_{\mathrm{null}}\)} is the secondary null-space torque, and
\mbox{\(\tau_c(q,\dot q)\)} is the Coriolis and centrifugal compensation obtained
from the robot model and added in the control callback. A separate gravity
term is not added because gravity compensation is handled internally by the
Franka Control Interface, as described in
\mbox{\Cref{subsec:coriolis_gravity_compensation}}.}
