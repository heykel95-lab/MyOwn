\chapter{Experiment 2: Virtual Wall Constraint}
\label{app:exp2}

\section{Overview}

Experiment~2 evaluates compliant interaction with a virtual plane.
The wall is defined by \(n_s=[0,0,1]^\top\), so the normal direction
coincides with the base-frame \(z\)-axis. To match the surface-frame
definition of \cref{eq:surface_task_frame_matrix}, choose
\(t_1=[1,0,0]^\top\) and \(t_2=[0,1,0]^\top\), giving
\(R_{\mathrm{surface}}=[\,n_s\;t_1\;t_2\,]\).

\textbf{Why the base-frame matrices are diagonal for this specific case:}
Because \(n_s\), \(t_1\), and \(t_2\) coincide with base-frame axes, the
transformed matrix remains diagonal even though the surface-frame order is
normal first. The base-frame stiffness is
\(K_{p,0}=R_{\mathrm{surface}}\operatorname{diag}(K_n,K_{t_1},K_{t_2})
R_{\mathrm{surface}}^\top=\operatorname{diag}(K_{t_1},K_{t_2},K_n)\).
For any tilted wall (\(R_{\mathrm{surface}}\neq\mathbf{I}_3\)), the same procedure
produces a full $3\times3$ \abbr{SPD} matrix with non-zero off-diagonals.

\section{C++ Implementation}

\begin{lstlisting}[style=cppstyle,
  caption={Experiment 2: Virtual wall controller with CSV logging},
  label={lst:exp2}]
// Experiment 2: Virtual Wall (n_s = [0,0,1])
// Surface-frame order: [normal, tangent_1, tangent_2]
// In-plane: k_free=50 N/m;  Normal: k_stiff=1000 N/m
// Tilt free (t1,t2): k_R_free=5 Nm/rad;  Spin stiff (n_s): k_R_stiff=200 Nm/rad

// Surface frame R_surface = [n_s, t1, t2].
Eigen::Vector3d n_s(0.0, 0.0, 1.0);
Eigen::Vector3d t1(1.0, 0.0, 0.0);
Eigen::Vector3d t2(0.0, 1.0, 0.0);
Eigen::Matrix3d R_surface;
R_surface.col(0) = n_s;
R_surface.col(1) = t1;
R_surface.col(2) = t2;

// Stiffness in surface frame (normal first, then tangents).
Eigen::Matrix3d Kp_task, KR_task;
Kp_task << 1000, 0,  0,
              0, 50, 0,
              0, 0, 50;   // [N/m]
KR_task << 200, 0, 0,
              0, 5, 0,
              0, 0, 5;   // [Nm/rad]

// Rotate to base frame: Kp = R_surface * Kp_task * R_surface^T.
// For this axis-aligned wall the result is diagonal in the base frame;
// for a tilted wall it becomes a full SPD matrix.
Eigen::Matrix3d Kp = R_surface * Kp_task * R_surface.transpose();
Eigen::Matrix3d KR = R_surface * KR_task * R_surface.transpose();

// Damping values selected per principal direction
Eigen::Matrix3d Dp = R_surface *
    Eigen::Vector3d(63.25, 14.14, 14.14)
    .asDiagonal() * R_surface.transpose();
Eigen::Matrix3d DR = R_surface *
    Eigen::Vector3d(28.28, 4.47, 4.47)
    .asDiagonal() * R_surface.transpose();

// Control law (same as Experiment 1):
Eigen::Vector3d pdot_d = Eigen::Vector3d::Zero();
Eigen::Vector3d f = Kp * e_p + Dp * (pdot_d - pdot);
Eigen::Vector3d m = KR * e_R - DR * omega;
Eigen::Matrix<double,6,1> F; F.head<3>()=f; F.tail<3>()=m;
Eigen::Matrix<double,7,1> tau = J.transpose() * F;
tau += tau_g + tau_c;

// Log wall-specific quantities to CSV
double d_wall = n_hat.dot(p_EE - p0) * 1000.0;  // [mm]
double phi_tilt = std::sqrt(e_R(0)*e_R(0)+e_R(1)*e_R(1)) * 180.0/M_PI;
double phi_spin = std::abs(e_R(2)) * 180.0/M_PI;
csv << time << "," << d_wall << "," << phi_tilt << "," << phi_spin;
for (int i=0; i<7; ++i) csv << "," << tau(i);
csv << "\n";
\end{lstlisting}

\section{Python Simulation Code}

The following (from \texttt{thesis\_plots.py}) simulates all three trials:

\begin{lstlisting}[style=cppstyle, language=Python,
  caption={Python simulation for Experiment~2 (virtual wall trials)},
  label={lst:exp2_sim}]
def simulate_sdof(K, D, M, F_fn, dt=0.001, T=4.0):
    t = np.arange(0, T, dt)
    x, xd = np.zeros(len(t)), np.zeros(len(t))
    for i in range(1, len(t)):
        xdd = (F_fn(t[i-1]) - D*xd[i-1] - K*x[i-1]) / M
        xd[i] = xd[i-1] + dt*xdd
        x[i]  = x[i-1]  + dt*xd[i-1]
    return t, x

# Trial A: 10 N normal push (k=1000 N/m, stiff)
F_A = lambda t: 10.0 if 0.5 < t < 1.5 else 0.0
t_A, d_n = simulate_sdof(1000, 2*np.sqrt(1000), 1.0, F_A)

# Trial B: 2 N in-plane slide (k=50 N/m, free)
F_B = lambda t: 2.0 if t > 0.5 else 0.0
t_B, d_t = simulate_sdof(50, 2*np.sqrt(50), 1.0, F_B)

# Trial C: tilt-to-contact (k_tilt=5 Nm/rad free, k_spin=200 stiff)
tau_tilt = lambda t: 0.5 if 0.5 < t < 2.5 else 0.0
t_C, phi_tilt = simulate_sdof(5, 2*np.sqrt(5), 1.0, tau_tilt)
\end{lstlisting}
