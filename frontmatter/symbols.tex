\renewcommand{\arraystretch}{1.2}
\newcommand{\symrow}[3]{#1 & #2 & #3\\}
% A group heading is useless without rows under it, so reserve room for the
% heading and the first few rows before placing it. \nobreak alone does not
% hold across a longtable, which begins its own breakable structure.
\newcommand{\symgroup}[2]{%
  \needspace{5\baselineskip}%
  \par\noindent\textbf{#1}\par\nobreak
  \begingroup
  \setlength{\LTpre}{2pt}%
  \setlength{\LTpost}{0.9em}%
  \begin{longtable}{@{}p{3.9cm}p{6.5cm}p{4.2cm}@{}}%
  #2
  \end{longtable}%
  \endgroup}

\symgroup{Frames and Kinematics}{
\symrow{$\{0\}$}{Robot base frame}{[-]}
\symrow{$\{\mathrm{EE}\}$}{End-effector frame used by the robot model}{[-]}
\symrow{$\{d\}$}{Desired end-effector and TCP reference frame}{[-]}
\symrow{$\{S\}$}{Surface task frame, with axes $t_1$, $t_2$ and $n_s$}{[-]}
\symrow{$n$}{Number of joints ($n=7$)}{[-]}
\symrow{$q,\ \dot{q}$}{Joint position / velocity vector}{[rad], [rad/s]}
\symrow{$p$}{Position / translation vector}{[m]}
\symrow{$R$}{Rotation matrix}{[-]}
\symrow{$T$}{Homogeneous transformation matrix}{[-]}
\symrow{$J(q)$}{Geometric Jacobian; linear rows above angular rows}{[m/rad], [-]}
}

\symgroup{Pose and Error}{
\symrow{$p_{\mathrm{EE}},\ p_{\mathrm{TCP}}$}{End-effector / tool-centre-point position; $p_{\mathrm{TCP}}=p_{\mathrm{EE}}$ in the implemented model}{[m]}
\symrow{$R_{\mathrm{EE}},\ T_{\mathrm{EE}}$}{End-effector orientation / transformation}{[-]}
\symrow{$p_d,\ R_d,\ T_d$}{Desired position / orientation / transformation}{[m], [-]}
\symrow{$\Delta R$}{Current-to-desired relative rotation, $R_{\mathrm{EE}}^\top R_d$}{[-]}
\symrow{$\phi$}{Orientation-error angle}{[rad]}
\symrow{$u,\ [u]_\times$}{Current-EE-frame orientation-error axis / its skew matrix}{[-]}
\symrow{$e_p$}{Translational (position) error}{[m]}
\symrow{$e_R$}{Base-frame rotational error used by the command}{[rad]}
\symrow{$\dot{x}$}{Cartesian velocity (twist)}{[m/s], [rad/s]}
}

\symgroup{Cartesian Impedance Gains}{
\symrow{$K_p,\ D_p$}{Generic translational stiffness / damping block; a frame or phase index is added when needed}{[N/m], [N\,s/m]}
\symrow{$K_R,\ D_R$}{Generic rotational stiffness / damping block; a frame or phase index is added when needed}{[N\,m/rad], [N\,m\,s/rad]}
\symrow{$K_{p,t_1},\ K_{p,t_2},\ K_{p,n}$}{Translational stiffness entries along $t_1$, $t_2$, and $n_s$}{[N/m]}
\symrow{$K_{R,t_1},\ K_{R,t_2},\ K_{R,n}$}{Rotational stiffness entries about $t_1$, $t_2$, and $n_s$}{[N\,m/rad]}
}

\symgroup{Dynamics and Torques}{
\symrow{$F$}{Cartesian wrench}{[N], [N\,m]}
\symrow{$f,\ m$}{Complete commanded Cartesian force / moment at the TCP, with $m=m_R+r_c\times f$}{[N], [N\,m]}
\symrow{$f_n,\ f_t$}{Normal / tangential component of the complete commanded force, $f_n=F_n n_s$ and $f_t=f-f_n$}{[N], [N]}
\symrow{$K,\ D$}{Full Cartesian stiffness / damping matrices}{$K$: [N/m], [N\,m/rad]; $D$: [N\,s/m], [N\,m\,s/rad]}
\symrow{$\tau$}{Joint torque vector}{[N\,m]}
\symrow{$\tau_c,\ \tau_g$}{Coriolis / gravity compensation torque}{[N\,m]}
\symrow{$\tau_{\mathrm{cart}}$}{Cartesian impedance torque, $J^\top(q)F$}{[N\,m]}
\symrow{$\tau_{\mathrm{cmd}}$}{Complete commanded joint torque}{[N\,m]}
}

\symgroup{Surface and Contact Geometry}{
\symrow{$n_s$}{Calibrated surface-plane normal copied into the controller and used in the evaluation}{[-]}
\symrow{$t_1,\ t_2$}{Surface tangent directions (unit)}{[-]}
\symrow{$R_{\mathrm{surface}}$}{Surface task frame $[\,t_1\;\;t_2\;\;n_s\,]$ in the base frame}{[-]}
\symrow{$n_{\mathrm{Tool,EE}},\ n_{\mathrm{Tool},0},\ n_d$}{Tool-face normal expressed in the EE frame, fixed by calibration / the same direction expressed in the base frame, $n_{\mathrm{Tool},0}=R_{\mathrm{EE}}n_{\mathrm{Tool,EE}}$ / signed controller target direction}{[-]}
\symrow{$\theta_{\mathrm{tilt}}$}{Commanded orientation-offset rotation vector, $\theta_{\mathrm{tilt}}=\theta_{t_1}t_1+\theta_{t_2}t_2$}{[rad]}
\symrow{$p_s$}{Point on the configured surface}{[m]}
\symrow{$p_c$}{Centre of compliance}{[m]}
\symrow{$r_c$}{TCP-to-compliance-centre lever, $p_c-p_{\mathrm{TCP}}$, as it enters the point-shift transformation and as the experimental plots and tables report it}{[m]}
\symrow{$r_{c,t_1},\ r_{c,t_2},\ r_{c,n}$}{Surface-frame components of $r_c$ along $t_1$, $t_2$ and $n_s$}{[m]}
\symrow{$r_{\mathrm{Tool,EE}},\ p_{\mathrm{Tool}}$}{Selected tool-point offset from the TCP, expressed in the EE frame / selected tool-point position in the base frame}{[m]}
\symrow{$r_{\mathrm{Tool}}$}{Selected tool-point offset from the TCP, $p_{\mathrm{Tool}}-p_{\mathrm{TCP}}$}{[m]}
\symrow{$m_R$}{Rotational-impedance contribution to the commanded moment, $K_Re_R-D_R\omega_{\mathrm{EE}}$, present at every lever setting}{[N\,m]}
}

\symgroup{Compliance and Alignment}{
\symrow{$\mathrm{Ad}(r_c)$}{TCP-to-compliance-centre point-shift adjoint}{[-]}
\symrow{$K_c,D_c$}{Block-diagonal stiffness / damping at the centre of compliance}{$K$: [N/m], [N\,m/rad]; $D$: [N\,s/m], [N\,m\,s/rad]}
\symrow{$F_c,\ F_{\mathrm{TCP}}$}{Cartesian wrench at the centre of compliance / corresponding wrench at the TCP}{[N], [N\,m]}
\symrow{$K_{\mathrm{TCP}},D_{\mathrm{TCP}}$}{Centre-of-compliance gains mapped by congruence to the TCP}{$K$: [N/m], [N\,m/rad]; $D$: [N\,s/m], [N\,m\,s/rad]}
\symrow{$\theta_{t_1},\ \theta_{t_2}$}{Surface-frame components of the commanded tool orientation offset}{[rad]}
\symrow{$\Delta\theta_i$}{Measured signed end-effector rotation about surface tangent $t_i$ from the beginning to the end of set-up}{[rad]}
\symrow{$r_{c,t}$}{Tangential part of the compliance-centre displacement, $r_{c,t_1}t_1+r_{c,t_2}t_2$; its magnitude is written $\lVert r_{c,t}\rVert$}{[m]}
\symrow{$F_n,\ M_{t_i}$}{Signed commanded normal force / commanded TCP moment about surface tangent $t_i$}{[N], [N\,m]}
}

\symgroup{Null-Space and Singular Value Decomposition}{
\symrow{$J^+$}{Thresholded-SVD Moore--Penrose pseudoinverse}{[rad/m], [-]}
\symrow{$N_q,\ N_\tau$}{Joint-velocity / torque null-space projector}{[-]}
\symrow{$\sigma_i,\ \sigma_{\min}$}{Jacobian singular value / $\sigma_6$ conditioning indicator. The geometric Jacobian mixes linear and angular rows, so a singular value carries no single physical unit and its numerical value depends on the chosen Jacobian representation and length unit; it is used as a relative indicator under one fixed convention}{[m/rad], [-]}
\symrow{$v_7$}{Structural null direction of the seven-joint Jacobian}{[-]}
\symrow{$\dot q_{\mathrm{null}}$}{Projected joint velocity in the numerical null space}{[rad/s]}
\symrow{$d_{\mathrm{null}}$}{Null-space damping}{[N\,m\,s/rad]}
\symrow{$k_\sigma$}{Sigma-conditioning torque magnitude}{[N\,m]}
\symrow{$\alpha_{\mathrm{probe}}$}{Null-direction sampling angle}{[rad]}
\symrow{$\Delta\sigma_{\min,\mathrm{dist}}$}{Experimental change in $\sigma_{\min}$ over the disturbance interval, $\sigma_{\min}(9\,\mathrm{s})-\sigma_{\min}(5\,\mathrm{s})$; distinct from the controller's $\Delta_\sigma$}{[m/rad], [-]}
\symrow{$\tau_d,\ \tau_\sigma,\ \tau_{\mathrm{null}}$}{Projected damping / sigma / total null-space torque}{[N\,m]}
\symrow{$E_N$}{Cumulative projected null-space motion; integrated projected joint-velocity magnitude during the disturbance interval, reported in degrees}{[rad]}
\symrow{$v_{\mathrm{ref}}$}{Fixed redundant comparison direction, taken from the condition without null-space torque}{[-]}
\symrow{$\Delta q_{\mathrm{null}},\ \Delta\eta_{\mathrm{dist}}$}{Net projected joint displacement over the disturbance interval / its signed component along $v_{\mathrm{ref}}$}{[rad]}
}
