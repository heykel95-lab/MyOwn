\addcontentsline{toc}{chapter}{List of Symbols}
\markboth{List of Symbols}{List of Symbols}
\ReviewMark{green}{The notation is extensive but well organised, and the unit
column makes the mixed translational and rotational quantities traceable.}

\Added{Cartesian quantities are expressed in the robot base frame
\mbox{\(\{0\}\)} unless a frame suffix says otherwise, and the base-frame
index is omitted. A suffix \mbox{\(S\)} or \mbox{\(\mathrm{EE}\)} after a
comma names the surface or end-effector frame the components are represented
in. An index that names a point or a body, as in
\mbox{\(p_{\mathrm{EE}}\)}, identifies that point rather than a coordinate
frame.}

\newcommand{\symrow}[3]{#1 & #2 & #3\\}
% A group heading is useless without rows under it, so reserve room for the
% heading and its first row before placing it. \nobreak alone does not
% hold across a longtable, which begins its own breakable structure.
\newcommand{\symgroup}[2]{%
  \needspace{3\baselineskip}%
  \par\noindent\textbf{#1}\par\nobreak
  \begingroup
  \setlength{\LTpre}{2pt}%
  \setlength{\LTpost}{0.9em}%
  \begin{longtable}{@{}
  >{\raggedright\arraybackslash}p{3.9cm}
  >{\raggedright\arraybackslash}p{6.5cm}
  >{\raggedright\arraybackslash}p{4.2cm}@{}}%
  #2
  \end{longtable}%
  \endgroup}

\symgroup{Frames and Kinematics}{
\symrow{$\{0\}$}{Robot base frame}{[-]}
\symrow{$\{\mathrm{EE}\}$}{End-effector frame used by the robot model}{[-]}
\symrow{$\{d\}$}{Desired end-effector and TCP reference frame}{[-]}
\symrow{$\{S\}$}{Surface frame, with axes $t_1$, $t_2$ and $n_s$}{[-]}
\symrow{$q,\ \dot{q}$}{Joint position / velocity vector}{[rad], [rad/s]}
\symrow{$q_{\mathrm{init}}$}{Configured initial joint posture}{[rad]}
\symrow{$p$}{Position / translation vector}{[m]}
\symrow{$R$}{Rotation matrix}{[-]}
\symrow{$T$}{Homogeneous transformation matrix}{Rotation block: [-]; translation block: [m]}
\symrow{$J(q)$}{Geometric Jacobian; linear rows above angular rows}{[m/rad], [-]}
}

\symgroup{Pose and Error}{
\symrow{$p_{\mathrm{EE}},\ p_{\mathrm{TCP}}$}{End-effector / tool-centre-point position; $p_{\mathrm{TCP}}=p_{\mathrm{EE}}$ in the implemented model}{[m]}
\symrow{$R_{\mathrm{EE}},\ T_{\mathrm{EE}}$}{End-effector orientation / transformation}{$R$: [-]; $T$: rotation block [-], translation block [m]}
\symrow{$R_{\mathrm{EE,clearance}}$, $p_{\mathrm{TCP}}(t_{\mathrm{start}})$}{End-effector orientation captured at the clearance transition and held as the contact-establishment orientation reference / TCP position captured at the same instant}{[-], [m]}
\symrow{$p_d,\ R_d,\ T_d$}{Desired position / orientation / transformation}{$p$: [m]; $R$: [-]; $T$: rotation block [-], translation block [m]}
\symrow{$\Delta R$}{Current-to-desired relative rotation in the base frame, $R_dR_{\mathrm{EE}}^\top$}{[-]}
\symrow{$\phi$}{Orientation-error angle}{[rad]}
\symrow{$u,\ [u]_\times$}{Base-frame orientation-error axis / its skew matrix}{[-]}
\symrow{$e_p$}{Translational (position) error}{[m]}
\symrow{$e_R$}{Base-frame rotational error used by the command}{[rad]}
\symrow{$\dot p_d,\ \omega_d$}{Desired end-effector linear / angular velocity; $\omega_d=0$ in the implemented controller}{[m/s], [rad/s]}
\symrow{$\dot p_{\mathrm{EE}},\ \omega_{\mathrm{EE}}$}{Measured end-effector linear / angular velocity}{[m/s], [rad/s]}
}

\symgroup{Cartesian Stiffness and Damping}{
\symrow{$K_p,\ D_p$}{Generic translational stiffness / damping block; a frame or state index is added when needed}{[N/m], [N\,s/m]}
\symrow{$K_R,\ D_R$}{Generic rotational stiffness / damping block; a frame or state index is added when needed}{[N\,m/rad], [N\,m\,s/rad]}
\symrow{$K_{p,t_1},\ K_{p,t_2},\ K_{p,n}$}{Translational stiffness entries along $t_1$, $t_2$, and $n_s$}{[N/m]}
\symrow{$K_{R,t_1},\ K_{R,t_2},\ K_{R,n}$}{Rotational stiffness entries about $t_1$, $t_2$, and $n_s$}{[N\,m/rad]}
\symrow{$D_{p,t_1},\ D_{p,t_2},\ D_{p,n}$}{Translational damping entries along $t_1$, $t_2$, and $n_s$}{[N\,s/m]}
\symrow{$D_{R,t_1},\ D_{R,t_2},\ D_{R,n}$}{Rotational damping entries about $t_1$, $t_2$, and $n_s$}{[N\,m\,s/rad]}
\symrow{$K,\ D$}{Complete base-frame Cartesian stiffness / damping, assembled block diagonal from the rotated translational and rotational blocks}{[N/m], [N\,m/rad]; [N\,s/m], [N\,m\,s/rad]}
}

\symgroup{Dynamics and Torques}{
\symrow{$M(q)$}{Joint-space inertia matrix}{[kg\,m$^2$]}
\symrow{$\Lambda(q),\ \Lambda_S(q)$}{Regularised operational-space inertia in the base frame / resolved along the surface-frame directions}{Translational: [kg]; coupling: [kg\,m]; rotational: [kg\,m$^2$]}
\symrow{$\lambda_i(q)$}{Directional Cartesian inertia, the $i$th diagonal entry of $\Lambda_S(q)$}{Translational: [kg]; rotational: [kg\,m$^2$]}
\symrow{$k_i,\ d_i$}{Directional stiffness / inertia-scaled damping coefficient}{Translational: [N/m], [N\,s/m]; rotational: [N\,m/rad], [N\,m\,s/rad]}
\symrow{$\zeta$}{Selected damping factor}{[-]}
\symrow{$F$}{Complete commanded Cartesian wrench}{[N], [N\,m]}
\symrow{$f,\ m$}{Complete commanded Cartesian force / moment at the TCP}{[N], [N\,m]}
\symrow{$F_n,\ f_n,\ f_t$}{Commanded normal-force component, $F_n=n_s^\top f$ / normal force vector, $f_n=F_nn_s$ / tangential force vector, $f_t=f-f_n$}{[N]}
\symrow{$M_{t_1}$}{Commanded TCP moment component about $t_1$, $M_{t_1}=t_1^\top m$}{[N\,m]}
\symrow{$\tau$}{Joint torque vector}{[N\,m]}
\symrow{$\tau_c$}{Coriolis and centrifugal compensation torque}{[N\,m]}
\symrow{$\tau_{\mathrm{cart}}$}{Cartesian impedance torque, $J^\top(q)F$}{[N\,m]}
\symrow{$\tau_{\mathrm{cmd}}$}{Complete commanded joint torque}{[N\,m]}
}

\symgroup{Surface and Contact Geometry}{
\symrow{$n_s$}{Configured surface-reference normal copied into the controller and used in the evaluation}{[-]}
\symrow{$n_{\mathrm{phys}}$}{Physical surface normal}{[-]}
\symrow{$t_1,\ t_2$}{Surface tangent directions (unit)}{[-]}
\symrow{$R_{\mathrm{surface}}$}{Surface-frame orientation $[\,t_1\;\;t_2\;\;n_s\,]$ in the base frame}{[-]}
\symrow{$n_{\mathrm{Tool,EE}},\ n_{\mathrm{Tool}},\ n_d$}{Tool-face normal expressed in the EE frame, obtained from calibration / the same direction expressed in the base frame, $n_{\mathrm{Tool}}=R_{\mathrm{EE}}n_{\mathrm{Tool,EE}}$ / desired tool-normal direction used by the controller}{[-]}
\symrow{$r_{\mathrm{face,EE}}$}{Calibrated tool-face-centre offset from the TCP in end-effector coordinates}{[m]}
\symrow{$\theta_{\mathrm{offset}}$}{Configured pre-contact orientation offset that sets the desired tool direction, $\theta_{\mathrm{offset}}=\theta_{\mathrm{offset},t_1}t_1+\theta_{\mathrm{offset},t_2}t_2$}{[rad]}
\symrow{$p_s$}{Point on the configured surface}{[m]}
\symrow{$e_{x_0},\ e_{z_0}$}{Unit vectors along the base-frame axes $x_0$ and $z_0$}{[-]}
\symrow{$p_c$}{Centre of compliance}{[m]}
\symrow{$r_c$}{TCP-to-compliance-centre displacement, $p_c-p_{\mathrm{TCP}}$, as the point-shift transformation and the experimental plots and tables use it}{[m]}
\symrow{$r_{c,S},\ r_{c,\mathrm{EE}}$}{The same displacement represented in the surface and end-effector frames, with $r_c=R_{\mathrm{surface}}r_{c,S}=R_{\mathrm{EE}}r_{c,\mathrm{EE}}$}{[m]}
\symrow{$r_{c,t_1},\ r_{c,t_2},\ r_{c,n}$}{Surface-frame components of $r_c$ along $t_1$, $t_2$ and $n_s$}{[m]}
\symrow{$r_{\mathrm{Tool,EE}},\ p_{\mathrm{Tool}}$}{Selected tool-point offset from the TCP, expressed in the EE frame / selected tool-point position in the base frame}{[m]}
\symrow{$p_{\mathrm{Tool}}(t_{\mathrm{start}})$, $p_{\mathrm{Tool},0}$, $p_{\mathrm{Tool},d}$}{Selected tool point at the start of Contact Establishment / its projection onto the configured surface reference / its desired contact-establishment position}{[m]}
\symrow{$s_a,\ v_a,\ s_{a,\max}$}{Commanded approach travel of the TCP along $-n_s$ / its configured rate / its configured maximum}{[m], [m/s], [m]}
\symrow{$p_{\mathrm{TCP,start}}$}{TCP position captured when surface approach begins}{[m]}
\symrow{$t_{\mathrm{start}},\ t_{\mathrm{end}}$}{Instants at which the state under discussion begins and ends; used for Surface Approach and for Contact Establishment}{[s]}
\symrow{$p_i$}{Base-frame position of tool-face corner $i$}{[m]}
\symrow{$h_i$}{Surface-normal clearance of tool-face corner $i$ from the configured surface reference}{[m]}
\symrow{$h_{\mathrm{Tool}}$}{Minimum clearance of the rectangular tool face from the configured surface reference, $h_{\mathrm{Tool}}=\min_i h_i$}{[m]}
\symrow{$h_{\mathrm{clearance}}$}{Configured tool clearance at which the transition from approach to contact establishment occurs}{[m]}
\symrow{$\varepsilon_{\mathrm{sel}}$}{Tool-point selection tolerance}{[m]}
\symrow{$I_{\mathrm{sel}},\ N$}{Set of tool-face corners within the selection tolerance of the minimum clearance / its size}{[-]}
\symrow{$\theta_{\mathrm{app,err}},\ \varepsilon_{\mathrm{app}}$}{Tool-orientation angular error during Tool Orientation / its configured tolerance}{[rad]}
\symrow{$s_c,\ v_c,\ s_{c,\max}$}{Commanded contact-establishment press coordinate of the selected tool point along $-n_s$ / its configured rate / its configured endpoint}{[m], [m/s], [m]}
\symrow{$m_R$}{Rotational-impedance contribution to the commanded moment, $K_Re_R-D_R\omega_{\mathrm{EE}}$, present at every compliance-centre position}{[N\,m]}

\symgroup{Compliance and Alignment}{
\symrow{$\mathrm{Ad}(r_c)$}{TCP-to-compliance-centre point-shift adjoint}{Diagonal blocks: [-]; point-shift block: [m/rad]}
\symrow{$K_c,D_c$}{Block-diagonal stiffness / damping at the centre of compliance}{$K$: [N/m], [N\,m/rad]; $D$: [N\,s/m], [N\,m\,s/rad]}
\symrow{$K_{\mathrm{TCP}},D_{\mathrm{TCP}}$}{Centre-of-compliance stiffness and damping mapped by congruence to the TCP}{$K$: [N/m], [N/rad], [N\,m/m], [N\,m/rad]; $D$: [N\,s/m], [N\,s/rad], [N\,m\,s/m], [N\,m\,s/rad]}
\symrow{$\theta_{\mathrm{offset},t_1},\ \theta_{\mathrm{offset},t_2}$}{Surface-frame components of the configured orientation offset about $t_1$ and $t_2$}{[rad]}
\symrow{$\phi_0,\ u_0$}{Magnitude and unit axis of the shortest rotation from the contact-entry tool normal to the inward configured surface normal; $\phi_0u_0$ is that rotation vector}{[rad], [-]}
\symrow{$\theta_{0,t_1}$}{Achieved pose-based initial angular offset about $t_1$, relative to the configured reference}{[rad]}
\symrow{$R_{\mathrm{CE}},\ u_{\mathrm{CE}},\ \phi_{\mathrm{CE}}$}{Relative rotation from the measured end orientation back to the held entry reference, and its axis and angle; the contact response is $\gamma=\phi_{\mathrm{CE}}u_{\mathrm{CE}}$}{[-], [-], [rad]}
\symrow{$\gamma,\ \gamma_{t_1}$}{Contact response vector / its component about $t_1$}{[rad]}
\symrow{$r_{c,t}$}{Tangential part of the compliance-centre displacement, $r_{c,t_1}t_1+r_{c,t_2}t_2$; its magnitude is written $\lVert r_{c,t}\rVert$}{[m]}
}

\symgroup{Null-Space and Singular Value Decomposition}{
\symrow{$J^+$}{Thresholded-SVD Moore--Penrose pseudoinverse}{[rad/m], [-]}
\symrow{$N_q,\ N_\tau$}{Joint-velocity / torque null-space projector}{[-]}
\symrow{$\sigma_i,\ \sigma_{\min}$}{Jacobian singular value / $\sigma_6$ conditioning indicator. The geometric Jacobian mixes linear and angular rows, so a singular value carries no single physical unit and its numerical value depends on the chosen Jacobian representation and length unit; it is used as a relative indicator under that convention}{Representation-dependent}
\symrow{$v_7$}{Structural null direction of the seven-joint Jacobian}{[-]}
\symrow{$\dot q_{\mathrm{null}}$}{Projected joint velocity in the numerical null space}{[rad/s]}
\symrow{$d_{\mathrm{null}}$}{Null-space damping}{[N\,m\,s/rad]}
\symrow{$k_\sigma$}{Sigma-conditioning torque magnitude}{[N\,m]}
\symrow{$\alpha_{\mathrm{probe}}$}{Null-direction sampling angle}{[rad]}
\symrow{$q_+,\ q_-$}{Joint configurations sampled either side of the current one along the null direction}{[rad]}
\symrow{$\sigma_+,\ \sigma_-,\ \Delta_\sigma$}{Minimum singular values at the two sampled configurations / their difference $\sigma_+-\sigma_-$}{Representation-dependent}
\symrow{$s_\sigma$}{Conditioning direction selector, $+1$, $-1$ or $0$ from $\Delta_\sigma$ against the deadband}{[-]}
\symrow{$\rho,\ \delta_\sigma$}{Relative SVD cutoff / sampled-singular-value comparison deadband}{[-], Representation-dependent}
\symrow{$\Delta\sigma_{\min}$}{Experimental change in $\sigma_{\min}$ over the disturbance interval, $\sigma_{\min}(9\,\mathrm{s})-\sigma_{\min}(5\,\mathrm{s})$; distinct from the controller's $\Delta_\sigma$}{Representation-dependent}
\symrow{$\tau_d,\ \tau_\sigma,\ \tau_{\mathrm{null}}$}{Projected damping / sigma / total null-space torque}{[N\,m]}
\symrow{$E_N$}{Cumulative projected null-space motion; integrated projected joint-velocity magnitude during the disturbance interval and converted to degrees for reporting}{[rad]}
\symrow{$u_f,\ F_{\mathrm{dist}},\ s_{\mathrm{dist}}$}{Fixed unit direction of the virtual disturbance / its commanded force magnitude / its commanded waveform in $[0,1]$}{[-], [N], [-]}
\symrow{$f_{\mathrm{dist}},\ \tau_{\mathrm{dist}}$}{Virtual disturbance force in the base frame / the equivalent joint torque, $J_p^\top f_{\mathrm{dist}}$}{[N], [N\,m]}
\symrow{$v_{\mathrm{ref}}$}{Redundant comparison direction, taken from the condition without null-space torque}{[-]}
\symrow{$\Delta q_{\mathrm{null}},\ \Delta\eta$}{Net projected joint displacement over the disturbance interval / its component along $v_{\mathrm{ref}}$, converted to degrees for reporting}{[rad]}
\symrow{$\lVert e_p\rVert_{\max}$}{Maximum Cartesian position error over the disturbance interval}{[m]}
}
