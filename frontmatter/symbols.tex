\renewcommand{\arraystretch}{1.2}
\newcommand{\symrow}[3]{#1 & #2 & #3\\}
% A group heading is useless without rows under it, so reserve room for the
% heading and the first few rows before placing it. \nobreak alone does not
% hold across a longtable, which begins its own breakable structure.
\newcommand{\symgroup}[2]{%
  \needspace{5\baselineskip}%
  \par\noindent\textbf{#1}\par\nobreak
  \begingroup
  \setlength{\LTpre}{2pt}%
  \setlength{\LTpost}{0.9em}%
  \begin{longtable}{@{}p{3.9cm}p{6.5cm}p{4.2cm}@{}}%
  #2
  \end{longtable}%
  \endgroup}

\symgroup{Frames and Kinematics}{
\symrow{$\{0\}$}{Robot base frame}{[-]}
\symrow{$\{\mathrm{EE}\}$}{End-effector frame used by the robot model}{[-]}
\symrow{$\{d\}$}{Desired end-effector and TCP reference frame}{[-]}
\symrow{$\{S\}$}{Surface frame, with axes $t_1$, $t_2$ and $n_s$}{[-]}
\symrow{$n$}{Number of joints ($n=7$)}{[-]}
\symrow{$q,\ \dot{q}$}{Joint position / velocity vector}{[rad], [rad/s]}
\symrow{$p$}{Position / translation vector}{[m]}
\symrow{$R$}{Rotation matrix}{[-]}
\symrow{$T$}{Homogeneous transformation matrix}{[-]}
\symrow{$J(q)$}{Geometric Jacobian; linear rows above angular rows}{[m/rad], [-]}
}

\symgroup{Pose and Error}{
\symrow{$p_{\mathrm{EE}},\ p_{\mathrm{TCP}}$}{End-effector / tool-centre-point position; $p_{\mathrm{TCP}}=p_{\mathrm{EE}}$ in the implemented model}{[m]}
\symrow{$R_{\mathrm{EE}},\ T_{\mathrm{EE}}$}{End-effector orientation / transformation}{[-]}
\symrow{$R_{\mathrm{EE,clearance}}$, $p_{\mathrm{TCP,clearance}}$}{End-effector orientation captured at the clearance transition and held as the contact-establishment orientation reference / TCP position captured at the same instant}{[-], [m]}
\symrow{$R_{\mathrm{CE,start}},\ R_{\mathrm{CE,end}}$}{Measured end-effector orientations at the beginning / end of contact establishment}{[-]}
\symrow{$R_{\mathrm{CE}}$}{Current-to-reference relative rotation over contact establishment, $R_{\mathrm{CE,start}}R_{\mathrm{CE,end}}^\top$}{[-]}
\symrow{$p_d,\ R_d,\ T_d$}{Desired position / orientation / transformation}{[m], [-]}
\symrow{$\Delta R_0$}{Current-to-desired relative rotation in the base frame, $R_dR_{\mathrm{EE}}^\top$}{[-]}
\symrow{$\phi$}{Orientation-error angle}{[rad]}
\symrow{$u,\ [u]_\times$}{Base-frame orientation-error axis / its skew matrix}{[-]}
\symrow{$e_p$}{Translational (position) error}{[m]}
\symrow{$e_R$}{Base-frame rotational error used by the command}{[rad]}
\symrow{$\dot{x}$}{Cartesian velocity (twist)}{[m/s], [rad/s]}
}

\symgroup{Cartesian Stiffness and Damping}{
\symrow{$K_p,\ D_p$}{Generic translational stiffness / damping block; a frame or state index is added when needed}{[N/m], [N\,s/m]}
\symrow{$K_R,\ D_R$}{Generic rotational stiffness / damping block; a frame or state index is added when needed}{[N\,m/rad], [N\,m\,s/rad]}
\symrow{$K_{p,t_1},\ K_{p,t_2},\ K_{p,n}$}{Translational stiffness entries along $t_1$, $t_2$, and $n_s$}{[N/m]}
\symrow{$K_{R,t_1},\ K_{R,t_2},\ K_{R,n}$}{Rotational stiffness entries about $t_1$, $t_2$, and $n_s$}{[N\,m/rad]}
\symrow{$D_{p,t_1},\ D_{p,t_2},\ D_{p,n}$}{Translational damping entries along $t_1$, $t_2$, and $n_s$}{[N\,s/m]}
\symrow{$D_{R,t_1},\ D_{R,t_2},\ D_{R,n}$}{Rotational damping entries about $t_1$, $t_2$, and $n_s$}{[N\,m\,s/rad]}
\symrow{$K_0,\ D_0$}{Complete base-frame Cartesian stiffness / damping, assembled block diagonal from the rotated translational and rotational blocks}{[N/m], [N\,m/rad]; [N\,s/m], [N\,m\,s/rad]}
}

\symgroup{Dynamics and Torques}{
\symrow{$M(q)$}{Joint-space inertia matrix}{[kg\,m$^2$]}
\symrow{$\Lambda_0(q),\ \Lambda_S(q)$}{Regularised operational-space inertia in base coordinates / resolved along the surface-frame directions}{Translational: [kg]; rotational: [kg\,m$^2$]}
\symrow{$T_R$}{Block rotation $\operatorname{diag}(R_{\mathrm{surface}},R_{\mathrm{surface}})$}{[-]}
\symrow{$\lambda_i(q)$}{Directional Cartesian inertia, the $i$th diagonal entry of $\Lambda_S(q)$}{Translational: [kg]; rotational: [kg\,m$^2$]}
\symrow{$k_i,\ d_i$}{Directional stiffness / inertia-scaled damping coefficient}{Translational: [N/m], [N\,s/m]; rotational: [N\,m/rad], [N\,m\,s/rad]}
\symrow{$\zeta,\ \varepsilon$}{Selected damping factor / operational-space-inertia regularisation constant}{[-], representation-dependent}
\symrow{$F$}{Cartesian wrench}{[N], [N\,m]}
\symrow{$f,\ m$}{Complete commanded Cartesian force / moment at the TCP}{[N], [N\,m]}
\symrow{$f_n,\ f_t$}{Normal / tangential component of the complete commanded force, $f_n=n_s(n_s^\top f)$ and $f_t=f-f_n$}{[N], [N]}
\symrow{$K,\ D$}{Full Cartesian stiffness / damping matrices}{$K$: [N/m], [N\,m/rad]; $D$: [N\,s/m], [N\,m\,s/rad]}
\symrow{$\tau$}{Joint torque vector}{[N\,m]}
\symrow{$\tau_c,\ \tau_g$}{Coriolis / gravity compensation torque}{[N\,m]}
\symrow{$\tau_{\mathrm{cart}}$}{Cartesian impedance torque, $J^\top(q)F$}{[N\,m]}
\symrow{$\tau_{\mathrm{cmd}}$}{Complete commanded joint torque}{[N\,m]}
}

\symgroup{Surface and Contact Geometry}{
\symrow{$n_s$}{Configured surface-reference normal copied into the controller and used in the evaluation}{[-]}
\symrow{$n_{\mathrm{phys}}$}{Physical surface normal; not measured independently in the reported experiments}{[-]}
\symrow{$t_1,\ t_2$}{Surface tangent directions (unit)}{[-]}
\symrow{$R_{\mathrm{surface}}$}{Surface-frame orientation $[\,t_1\;\;t_2\;\;n_s\,]$ in the base frame}{[-]}
\symrow{$n_{\mathrm{Tool,EE}},\ n_{\mathrm{Tool},0},\ n_d$}{Tool-face normal expressed in the EE frame, fixed by calibration / the same direction expressed in the base frame, $n_{\mathrm{Tool},0}=R_{\mathrm{EE}}n_{\mathrm{Tool,EE}}$ / desired tool-normal direction used by the controller}{[-]}
\symrow{$\theta_{\mathrm{cmd}}$}{Commanded tool-orientation offset relative to the configured surface reference, $\theta_{\mathrm{cmd}}=\theta_{t_1}t_1+\theta_{t_2}t_2$}{[rad]}
\symrow{$p_s$}{Point on the configured surface}{[m]}
\symrow{$e_{x_0},\ e_{z_0}$}{Unit vectors along the base-frame axes $x_0$ and $z_0$}{[-]}
\symrow{$p_c$}{Centre of compliance}{[m]}
\symrow{$r_c$}{TCP-to-compliance-centre displacement, $p_c-p_{\mathrm{TCP}}$, as the point-shift transformation and the experimental plots and tables use it; where no frame index is shown, the base-frame representation is implied}{[m]}
\symrow{$r_{c,0},\ r_{c,S},\ r_{c,\mathrm{EE}}$}{Representations of the same displacement in the base, surface and end-effector frames, with $r_{c,0}=R_{\mathrm{surface}}r_{c,S}=R_{\mathrm{EE}}r_{c,\mathrm{EE}}$}{[m]}
\symrow{$r_{c,t_1},\ r_{c,t_2},\ r_{c,n}$}{Surface-frame components of $r_c$ along $t_1$, $t_2$ and $n_s$}{[m]}
\symrow{$r_{\mathrm{Tool,EE}},\ p_{\mathrm{Tool}}$}{Selected tool-point offset from the TCP, expressed in the EE frame / selected tool-point position in the base frame}{[m]}
\symrow{$p_{\mathrm{Tool,clearance}}$, $p_{\mathrm{Tool},0}$, $p_{\mathrm{Tool},d}$}{Selected tool point at the clearance transition / its projection onto the configured surface reference / its desired contact-establishment position}{[m]}
\symrow{$s_{\mathrm{app}},\ v_{\mathrm{app}},\ s_{\mathrm{app,max}}$}{Commanded approach travel of the TCP along $-n_s$ / its configured rate / its configured maximum}{[m], [m/s], [m]}
\symrow{$p_{\mathrm{TCP,start}}$}{TCP position captured when surface approach begins}{[m]}
\symrow{$h_i$}{Signed clearance of tool-face corner $i$ from the configured surface reference}{[m]}
\symrow{$h_{\mathrm{Tool}}$}{Minimum clearance of the rectangular tool face from the configured surface reference, $h_{\mathrm{Tool}}=\min_i h_i$}{[m]}
\symrow{$h_{\mathrm{clearance}}$}{Configured tool clearance at which the transition from approach to contact establishment occurs}{[m]}
\symrow{$\varepsilon_{\mathrm{sel}}$}{Tool-point selection tolerance}{[m]}
\symrow{$s_{\mathrm{CE}},\ v_{\mathrm{CE}},\ s_{\mathrm{CE,max}}$}{Commanded contact-establishment press coordinate of the selected tool point along $-n_s$ / its configured rate / its configured endpoint}{[m], [m/s], [m]}
\symrow{$t_{\mathrm{CE,start}},\ t_{\mathrm{CE,end}}$}{Measurement instants at the beginning / end of Contact Establishment}{[s]}
\symrow{$m_R$}{Rotational-impedance contribution to the commanded moment, $K_Re_R-D_R\omega_{\mathrm{EE}}$, present at every compliance-centre position}{[N\,m]}
}

\symgroup{Compliance and Alignment}{
\symrow{$\mathrm{Ad}(r_c)$}{TCP-to-compliance-centre point-shift adjoint}{[-]}
\symrow{$K_c,D_c$}{Block-diagonal stiffness / damping at the centre of compliance}{$K$: [N/m], [N\,m/rad]; $D$: [N\,s/m], [N\,m\,s/rad]}
\symrow{$K_{\mathrm{TCP}},D_{\mathrm{TCP}}$}{Centre-of-compliance stiffness and damping mapped by congruence to the TCP}{$K$: [N/m], [N\,m/rad]; $D$: [N\,s/m], [N\,m\,s/rad]}
\symrow{$\theta_{t_1},\ \theta_{t_2}$}{Surface-frame components of the commanded tool orientation offset}{[rad]}
\symrow{$\theta_{0,S}$}{Pose-based initial angular-deviation vector in surface coordinates}{[rad]}
\symrow{$\theta_{0,t_1},\ \theta_{0,t_2}$}{Achieved pose-based initial angular deviation about the surface tangents, relative to the configured reference}{[rad]}
\symrow{$\gamma_0,\ \gamma_S$}{Signed contact-establishment response vector in base / surface coordinates: the end-of-contact-establishment orientation change relative to the held contact-establishment reference}{[rad], [rad]}
\symrow{$\gamma_{t_1},\ \gamma_{t_2},\ \gamma_n$}{Surface-frame components of the signed contact-establishment response; a component reducing an initial deviation carries the same sign as that deviation}{[rad]}
\symrow{$r_{c,t}$}{Tangential part of the compliance-centre displacement, $r_{c,t_1}t_1+r_{c,t_2}t_2$; its magnitude is written $\lVert r_{c,t}\rVert$}{[m]}
}

\symgroup{Null-Space and Singular Value Decomposition}{
\symrow{$J^+$}{Thresholded-SVD Moore--Penrose pseudoinverse}{[rad/m], [-]}
\symrow{$N_q,\ N_\tau$}{Joint-velocity / torque null-space projector}{[-]}
\symrow{$\sigma_i,\ \sigma_{\min}$}{Jacobian singular value / $\sigma_6$ conditioning indicator. The geometric Jacobian mixes linear and angular rows, so a singular value carries no single physical unit and its numerical value depends on the chosen Jacobian representation and length unit; it is used as a relative indicator under one fixed convention}{[-]}
\symrow{$v_7$}{Structural null direction of the seven-joint Jacobian}{[-]}
\symrow{$\dot q_{\mathrm{null}}$}{Projected joint velocity in the numerical null space}{[rad/s]}
\symrow{$d_{\mathrm{null}}$}{Null-space damping}{[N\,m\,s/rad]}
\symrow{$k_\sigma$}{Sigma-conditioning torque magnitude}{[N\,m]}
\symrow{$\alpha_{\mathrm{probe}}$}{Null-direction sampling angle}{[rad]}
\symrow{$\Delta\sigma_{\min,\mathrm{dist}}$}{Experimental change in $\sigma_{\min}$ over the disturbance interval, $\sigma_{\min}(9\,\mathrm{s})-\sigma_{\min}(5\,\mathrm{s})$; distinct from the controller's $\Delta_\sigma$}{[-]}
\symrow{$\tau_d,\ \tau_\sigma,\ \tau_{\mathrm{null}}$}{Projected damping / sigma / total null-space torque}{[N\,m]}
\symrow{$E_N$}{Cumulative projected null-space motion; integrated projected joint-velocity magnitude during the disturbance interval, converted to degrees for reporting}{[${}^\circ$]}
\symrow{$v_{\mathrm{ref}}$}{Fixed redundant comparison direction, taken from the condition without null-space torque}{[-]}
\symrow{$\Delta q_{\mathrm{null}},\ \Delta\eta_{\mathrm{dist}}$}{Net projected joint displacement over the disturbance interval / its signed component along $v_{\mathrm{ref}}$}{[rad]}
}
